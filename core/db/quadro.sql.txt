-- =====================================================================
--  1. O ROTEIRO (modelo unico, compartilhado por todas as obras)
--
--  etapa        -> a coluna vertical do quadro ("2a Etapa")
--  etapa_card   -> o bloco dentro da etapa; pertence a 1..N cargos
--                  (mais de um cargo = o card ganha gradiente na tela)
--  etapa_check  -> cada linha marcavel dentro do card
-- =====================================================================

CREATE TABLE IF NOT EXISTS etapa (
    id         BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    ordem      SMALLINT    NOT NULL,
    nome       TEXT        NOT NULL,
    criado_em  TIMESTAMPTZ NOT NULL DEFAULT now()
);

COMMENT ON COLUMN etapa.ordem IS 'Posicao no quadro. O rotulo ("3a Etapa") sai daqui.';

CREATE TABLE IF NOT EXISTS etapa_card (
    id        BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    etapa_id  BIGINT   NOT NULL REFERENCES etapa (id) ON DELETE CASCADE,
    ordem     SMALLINT NOT NULL DEFAULT 0,
    titulo    TEXT
);

COMMENT ON COLUMN etapa_card.titulo IS 'NULL = usa o nome do(s) cargo(s) do card.';

CREATE INDEX IF NOT EXISTS etapa_card_etapa_idx ON etapa_card (etapa_id, ordem);

-- um card pode responder por mais de um cargo; a tela pinta o gradiente
CREATE TABLE IF NOT EXISTS etapa_card_cargo (
    card_id  BIGINT   NOT NULL REFERENCES etapa_card (id) ON DELETE CASCADE,
    cargo_id BIGINT   NOT NULL REFERENCES cargo (id)      ON DELETE CASCADE,
    ordem    SMALLINT NOT NULL DEFAULT 0,
    PRIMARY KEY (card_id, cargo_id)
);

CREATE TABLE IF NOT EXISTS etapa_check (
    id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    card_id BIGINT   NOT NULL REFERENCES etapa_card (id) ON DELETE CASCADE,
    ordem   SMALLINT NOT NULL DEFAULT 0,
    titulo  TEXT     NOT NULL
);

CREATE INDEX IF NOT EXISTS etapa_check_card_idx ON etapa_check (card_id, ordem);


-- ---------------------------------------------------------------------
--  Dono de UM check
--
--  Normalmente o check e de quem e o card. Esta tabela e a excecao: um
--  check especifico pode responder a outro(s) cargo(s) sem que o card
--  inteiro mude de dono.
--
--  Exemplo real: "Envio revisao externa" mora no card do GQ, mas quem
--  marca sao GQ e Excelencia. Sem isto seria preciso ou criar um card
--  novo, ou transformar o card em "GQ + Excelencia" — e ai TODOS os
--  checks dele mudariam de dono junto.
--
--  Sem linha aqui = o check segue o card. Com linha, esta lista MANDA.
-- ---------------------------------------------------------------------

CREATE TABLE IF NOT EXISTS etapa_check_cargo (
    check_id BIGINT   NOT NULL REFERENCES etapa_check (id) ON DELETE CASCADE,
    cargo_id BIGINT   NOT NULL REFERENCES cargo (id)       ON DELETE CASCADE,
    ordem    SMALLINT NOT NULL DEFAULT 0,
    PRIMARY KEY (check_id, cargo_id)
);

COMMENT ON TABLE etapa_check_cargo IS
    'Dono de um check especifico. Sem linha, o check segue o cargo do card.';


-- =====================================================================
--  2. O QUE CADA OBRA JA MARCOU
--
--  Uma linha por check marcado. Nao ha linha para check aberto: a obra
--  nasce sem nenhuma e a tela cruza o roteiro com o que esta aqui.
--  Assim, mexer no roteiro (novo check, check apagado) vale na hora
--  para todas as obras, sem precisar reescrever obra por obra.
-- =====================================================================

CREATE TABLE IF NOT EXISTS obra_check (
    obra_id   BIGINT      NOT NULL REFERENCES obra (id)         ON DELETE CASCADE,
    check_id  BIGINT      NOT NULL REFERENCES etapa_check (id)  ON DELETE CASCADE,
    feito_por BIGINT      REFERENCES usuario (id)               ON DELETE SET NULL,
    feito_em  TIMESTAMPTZ NOT NULL DEFAULT now(),
    PRIMARY KEY (obra_id, check_id)
);

CREATE INDEX IF NOT EXISTS obra_check_obra_idx ON obra_check (obra_id);

-- quem marca um check entra como participante da obra (avatares do card)
CREATE OR REPLACE FUNCTION obra_check_entra() RETURNS trigger AS $$
BEGIN
    IF NEW.feito_por IS NOT NULL THEN
        INSERT INTO obra_membro (obra_id, usuario_id)
        VALUES (NEW.obra_id, NEW.feito_por)
        ON CONFLICT DO NOTHING;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS obra_check_entra_tg ON obra_check;
CREATE TRIGGER obra_check_entra_tg
    AFTER INSERT ON obra_check
    FOR EACH ROW EXECUTE FUNCTION obra_check_entra();


-- =====================================================================
--  3. OBSERVACOES DO QUADRO
--  As da tela de Obras, que valem para o quadro inteiro e nao para uma
--  obra so. As de dentro da obra continuam em obra_observacao.
-- =====================================================================

CREATE TABLE IF NOT EXISTS observacao_quadro (
    id          BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    usuario_id  BIGINT      REFERENCES usuario (id) ON DELETE SET NULL,
    autor_nome  TEXT        NOT NULL,
    texto       TEXT        NOT NULL,
    enviada_em  TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS observacao_quadro_data_idx ON observacao_quadro (enviada_em DESC);

-- =====================================================================
--  5. EMERGENCIA E SEMPRE PRIORIDADE ALTA
--  Vale para quem inserir pela tela e para quem inserir na mao.
-- =====================================================================

CREATE OR REPLACE FUNCTION obra_emergencia_alta() RETURNS trigger AS $$
BEGIN
    IF NEW.tipo = 'emergencia' THEN
        NEW.prioridade := 'alta';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS obra_emergencia_alta_tg ON obra;
CREATE TRIGGER obra_emergencia_alta_tg
    BEFORE INSERT OR UPDATE ON obra
    FOR EACH ROW EXECUTE FUNCTION obra_emergencia_alta();

UPDATE obra SET prioridade = 'alta' WHERE tipo = 'emergencia' AND prioridade <> 'alta';


-- =====================================================================
--  6. AS VIEWS PASSAM A OLHAR obra_check
--  (antes olhavam obra_tarefa, que este arquivo aposenta)
--
--  Obra concluida = nao sobrou nenhum check do roteiro sem marcar.
--  Obra sem nenhum check no roteiro nao conta como concluida.
-- =====================================================================

DROP VIEW IF EXISTS obra_conclusao;
CREATE VIEW obra_conclusao AS
SELECT o.id                                   AS obra_id,
       max(m.feito_em)                        AS concluida_em,
       count(*) FILTER (WHERE m.obra_id IS NULL) AS checks_abertos
  FROM obra o
 CROSS JOIN etapa_check ec
  LEFT JOIN obra_check m ON m.obra_id = o.id AND m.check_id = ec.id
 GROUP BY o.id
HAVING count(*) FILTER (WHERE m.obra_id IS NULL) = 0
   AND count(ec.id) > 0;

COMMENT ON VIEW obra_conclusao IS
    'Obras sem nenhum check em aberto. concluida_em = carimbo do ultimo check marcado.';

DROP VIEW IF EXISTS usuario_media;
CREATE VIEW usuario_media AS
SELECT u.id                  AS usuario_id,
       count(a.obra_id)      AS obras_avaliadas,
       round(avg(a.nota), 1) AS media
  FROM usuario u
  JOIN obra_membro    m ON m.usuario_id = u.id
  JOIN obra_avaliacao a ON a.obra_id    = m.obra_id
 GROUP BY u.id;

-- obra_tarefa foi substituida por etapa_check + obra_check
DROP TRIGGER IF EXISTS obra_membro_entra_tg ON obra_tarefa;
DROP TRIGGER IF EXISTS obra_tarefa_carimbo_tg ON obra_tarefa;
DROP TABLE IF EXISTS obra_tarefa;

-- =====================================================================
--  8. CONFERINDO
-- =====================================================================

-- o roteiro inteiro, do jeito que a tela mostra
SELECT e.ordem AS etapa, e.nome AS etapa_nome,
       string_agg(DISTINCT c.nome, ' + ' ORDER BY c.nome) AS cargos,
       ck.ordem AS n, ck.titulo AS "check"
  FROM etapa e
  JOIN etapa_card       ca ON ca.etapa_id = e.id
  LEFT JOIN etapa_card_cargo cc ON cc.card_id = ca.id
  LEFT JOIN cargo        c ON c.id = cc.cargo_id
  JOIN etapa_check      ck ON ck.card_id = ca.id
 GROUP BY e.ordem, e.nome, ca.ordem, ck.ordem, ck.titulo
 ORDER BY e.ordem, ca.ordem, ck.ordem;

-- quantos checks cada obra ja fechou
SELECT o.id, cl.nome AS cliente,
       count(m.check_id) AS feitos,
       (SELECT count(*) FROM etapa_check) AS total
  FROM obra o
  JOIN cliente cl ON cl.id = o.cliente_id
  LEFT JOIN obra_check m ON m.obra_id = o.id
 GROUP BY o.id, cl.nome
 ORDER BY o.id;
