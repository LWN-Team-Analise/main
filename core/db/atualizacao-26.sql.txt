-- ============================================================
--  ATUALIZACAO 26 — NOTIFICACOES NO CELULAR E MENSAGENS NAO LIDAS
--
--  1. push_inscricao — cada aparelho (celular, computador) em que a
--     pessoa ligou as notificacoes. O navegador entrega um endereco
--     (endpoint) e duas chaves; o servidor usa os tres para mandar a
--     notificacao. Uma pessoa pode ter varios aparelhos. Quando o
--     navegador diz que a inscricao morreu (a pessoa desinstalou o app
--     ou tirou a permissao), o servidor apaga a linha.
--
--  2. chat_site_leitura — ate qual mensagem do chat da equipe cada
--     pessoa ja leu. As mensagens de outras pessoas depois dela sao as
--     "nao lidas" do botao do chat.
--
--  SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon.
--
--  Rodar uma vez. Rodar de novo nao quebra nada e nao duplica nada.
-- ============================================================

BEGIN;

-- ---- 1. os aparelhos com notificacao ligada ----
CREATE TABLE IF NOT EXISTS public.push_inscricao (
    id          bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    usuario_id  bigint      NOT NULL REFERENCES public.usuario (id) ON DELETE CASCADE,
    endpoint    text        NOT NULL UNIQUE,
    p256dh      text        NOT NULL,
    auth        text        NOT NULL,
    aparelho    text,
    criado_em   timestamptz NOT NULL DEFAULT now(),
    usado_em    timestamptz
);

CREATE INDEX IF NOT EXISTS push_inscricao_do_usuario ON public.push_inscricao (usuario_id);

COMMENT ON TABLE public.push_inscricao IS
    'Aparelhos com notificacao no celular ligada (web push): o endereco e as chaves que o navegador entregou.';

-- ---- 2. ate onde cada um leu o chat da equipe ----
CREATE TABLE IF NOT EXISTS public.chat_site_leitura (
    usuario_id    bigint      PRIMARY KEY REFERENCES public.usuario (id) ON DELETE CASCADE,
    lido_ate      bigint      NOT NULL DEFAULT 0,
    atualizado_em timestamptz NOT NULL DEFAULT now()
);

COMMENT ON TABLE public.chat_site_leitura IS
    'Ate qual mensagem (chat_site.id) a pessoa ja leu o chat da equipe. As de outras pessoas depois dela sao as nao lidas.';

COMMIT;

-- ============================================================
--  CONFERENCIA — deve devolver "t" em todas as linhas
-- ============================================================

SELECT 'push_inscricao' AS item, to_regclass('public.push_inscricao') IS NOT NULL AS ok
UNION ALL
SELECT 'chat_site_leitura', to_regclass('public.chat_site_leitura') IS NOT NULL;
