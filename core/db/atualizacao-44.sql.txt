-- ============================================================
--  ATUALIZACAO 44 — A AGENDA DA EMPRESA (aba Panorama da pagina inicial)
--
--  A pagina inicial ganha a aba "Panorama", com quatro cards: avisos,
--  aniversariantes do mes, agenda da empresa e destaques das
--  avaliacoes. Tres deles leem o que o sistema ja tem; a agenda e a
--  unica coisa nova, e mora aqui:
--
--    agenda_evento   cada evento da empresa (reuniao geral,
--                    confraternizacao, treinamento): o titulo, o dia,
--                    a hora (opcional) e o local (opcional).
--
--  Ver a agenda e de todo mundo que entra no sistema. Criar, editar e
--  excluir evento e de quem tem `enviar_avisos` (o acesso total passa
--  por qualquer permissao): a agenda e um aviso com data marcada. Por
--  isso nao ha permissao nova.
--
--  SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon.
--
--  Rodar uma vez. Rodar de novo nao quebra nada e nao duplica nada.
-- ============================================================

BEGIN;

CREATE TABLE IF NOT EXISTS public.agenda_evento (
    id             bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    titulo         text        NOT NULL,
    dia            date        NOT NULL,
    hora           time,
    local          text,
    criado_por     bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    criado_em      timestamptz NOT NULL DEFAULT now(),
    atualizado_por bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    atualizado_em  timestamptz,
    CONSTRAINT agenda_evento_titulo_valido CHECK (btrim(titulo) <> '' AND length(titulo) <= 120),
    CONSTRAINT agenda_evento_local_valido CHECK (local IS NULL OR length(local) <= 120)
);

CREATE INDEX IF NOT EXISTS agenda_evento_dia_idx ON public.agenda_evento (dia, hora);

COMMENT ON TABLE public.agenda_evento IS
    'Eventos da empresa (aba Panorama da pagina inicial). Todos veem; quem tem enviar_avisos cria, edita e exclui.';
COMMENT ON COLUMN public.agenda_evento.hora IS
    'Hora de inicio. NULL = evento sem hora marcada (o dia inteiro).';
COMMENT ON COLUMN public.agenda_evento.local IS
    'Onde acontece (Sala de Reunioes, Online...). NULL = sem local.';

COMMIT;

-- ============================================================
--  CONFERENCIA — deve devolver "t" na linha
-- ============================================================

SELECT 'tabela agenda_evento' AS item,
       to_regclass('public.agenda_evento') IS NOT NULL AS ok;
