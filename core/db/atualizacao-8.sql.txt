-- ============================================================
--  ATUALIZACAO 8 — DESPESAS
--
--  A aba Despesas: cada pessoa envia as proprias DESPESAS
--  (com comprovante), REFEICOES (valor fixo) e BONUS, e acompanha
--  tudo em "Meus envios". Quem tem a permissao
--  `revisar_despesa_geral` ve os envios de toda a equipe.
--
--  Duas tabelas:
--
--    despesa_envio  um envio — despesa, refeicao OU bonus;
--    despesa_anexo  o comprovante de um envio.
--
--  Por que UMA tabela para as tres categorias, e nao tres:
--  elas tem exatamente a mesma forma (quem, obra, dia, tipo,
--  valor, observacao) e sao lidas SEMPRE juntas — "Meus envios"
--  lista as tres no mesmo dia e o total do mes e a soma das tres.
--  Em tres tabelas, cada lista e cada total viraria um UNION de
--  tres consultas. A `categoria` diz o que o envio e, e a trava
--  `despesa_envio_tipo_ok` garante que cada categoria so aceita
--  os tipos dela (refeicao nunca e "combustivel").
--
--  A obra e REFERENCIA, nao copia: o nome do cliente e o n° da
--  proposta saem da propria obra na hora de ler. O cliente nao
--  e gravado a parte porque ja e da obra (obra.cliente_id) — e a
--  obra pode trocar de cliente numa edicao, o que deixaria uma
--  copia mentindo.
--
--  Excluir a obra NAO apaga a despesa: registro de dinheiro fica.
--  O envio perde so o vinculo (obra_id vira NULL) e a tela passa
--  a dizer "obra excluída". O contrario — impedir a exclusao da
--  obra que tem despesa — travaria o "Excluir obra concluída" e o
--  excluir cliente, que ja existem.
--
--  Pessoa desligada no sistema e desativada (usuario.ativo), nao
--  apagada; por isso o usuario_id e RESTRICT: quem tem despesa
--  nao some do banco levando o historico junto.
--
--  O valor e NUMERIC(12,2), nunca texto: soma e total saem do
--  proprio banco, sem arredondamento de ponto flutuante.
--
--  Rodar uma vez. Tudo e IF NOT EXISTS: rodar de novo nao
--  quebra nada e nao apaga envio nenhum.
-- ============================================================

BEGIN;

-- ---- o envio ----
CREATE TABLE IF NOT EXISTS public.despesa_envio (
    id             bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    usuario_id     bigint NOT NULL REFERENCES public.usuario(id) ON DELETE RESTRICT,
    obra_id        bigint REFERENCES public.obra(id) ON DELETE SET NULL,
    categoria      text NOT NULL,
    tipo           text NOT NULL,
    data           date NOT NULL,
    valor          numeric(12,2) NOT NULL,
    justificativa  text,
    observacao     text,
    criado_em      timestamp with time zone NOT NULL DEFAULT now(),
    atualizado_em  timestamp with time zone NOT NULL DEFAULT now(),

    CONSTRAINT despesa_envio_categoria_ok CHECK (categoria IN ('despesa', 'refeicao', 'bonus')),
    CONSTRAINT despesa_envio_tipo_ok CHECK (
           (categoria = 'despesa'  AND tipo IN ('vale_transporte', 'transporte_intermunicipal', 'combustivel', 'outros'))
        OR (categoria = 'refeicao' AND tipo IN ('almoco', 'janta'))
        OR (categoria = 'bonus'    AND tipo IN ('bonus_viagem', 'bonus_apartamento'))
    ),
    CONSTRAINT despesa_envio_valor_ok CHECK (valor > 0),
    -- a justificativa e so do tipo "Outros"
    CONSTRAINT despesa_envio_justificativa_ok CHECK (justificativa IS NULL OR tipo = 'outros'),
    CONSTRAINT despesa_envio_justificativa_tam CHECK (justificativa IS NULL OR char_length(justificativa) <= 500),
    CONSTRAINT despesa_envio_observacao_tam CHECK (observacao IS NULL OR char_length(observacao) <= 1000)
);

-- "meus envios do mes" e a leitura de todo dia: pessoa + data
CREATE INDEX IF NOT EXISTS despesa_envio_usuario_data_idx
    ON public.despesa_envio (usuario_id, data);

-- a visao geral (todos os usuarios) corta so pela data
CREATE INDEX IF NOT EXISTS despesa_envio_data_idx
    ON public.despesa_envio (data);

-- sem este, excluir uma obra varreria a tabela inteira para o SET NULL
CREATE INDEX IF NOT EXISTS despesa_envio_obra_idx
    ON public.despesa_envio (obra_id);

-- o mesmo carimbo que a obra usa (db/sistema.sql.txt)
DROP TRIGGER IF EXISTS despesa_envio_atualizado_tg ON public.despesa_envio;
CREATE TRIGGER despesa_envio_atualizado_tg
    BEFORE UPDATE ON public.despesa_envio
    FOR EACH ROW EXECUTE FUNCTION toca_atualizado_em();

-- ---- o comprovante ----
--
-- Guardado como data URL, igual a obra_anexo e ao chat. O conteudo
-- nunca vem na lista: "Meus envios" traz nome e tamanho, e o
-- arquivo so e lido no clique de baixar.
CREATE TABLE IF NOT EXISTS public.despesa_anexo (
    id          bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    envio_id    bigint NOT NULL REFERENCES public.despesa_envio(id) ON DELETE CASCADE,
    nome        text NOT NULL,
    tipo        text,
    tamanho     integer,
    conteudo    text NOT NULL,
    enviado_em  timestamp with time zone NOT NULL DEFAULT now(),
    CONSTRAINT despesa_anexo_nome_tam CHECK (char_length(btrim(nome)) BETWEEN 1 AND 255),
    CONSTRAINT despesa_anexo_data_url CHECK (conteudo LIKE 'data:%')
);

CREATE INDEX IF NOT EXISTS despesa_anexo_envio_idx
    ON public.despesa_anexo (envio_id);

COMMENT ON TABLE public.despesa_envio IS
    'Despesas, refeicoes e bonus enviados pela equipe. categoria + tipo dizem o que e.';
COMMENT ON COLUMN public.despesa_envio.valor IS
    'Valor em reais do dia do envio. Na refeicao e o valor fixo de src/domain/despesas.js.';
COMMENT ON COLUMN public.despesa_envio.obra_id IS
    'A obra (e por ela o cliente). NULL = a obra foi excluida depois do envio.';
COMMENT ON TABLE public.despesa_anexo IS
    'Comprovante do envio, em data URL (base64). Teto de 3 MB por arquivo, conferido na API.';

-- ---- a permissao nova ----
--
-- O setor com acesso total ja passa por ela sem estar na lista;
-- ela entra la so para a tela de Setores mostrar a caixa marcada,
-- do mesmo jeito que foi feito com `excluir_concluidas`.
UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['revisar_despesa_geral']
 WHERE acesso_total
   AND NOT ('revisar_despesa_geral' = ANY (permissoes));

COMMIT;

-- ============================================================
--  CONFERENCIA — deve devolver "t" nas tres linhas
-- ============================================================

SELECT 'despesa_envio' AS item, to_regclass('public.despesa_envio') IS NOT NULL AS ok
UNION ALL SELECT 'despesa_anexo', to_regclass('public.despesa_anexo') IS NOT NULL
UNION ALL SELECT 'gatilho despesa_envio_atualizado_tg',
                 EXISTS (SELECT 1 FROM pg_trigger WHERE tgname = 'despesa_envio_atualizado_tg');
