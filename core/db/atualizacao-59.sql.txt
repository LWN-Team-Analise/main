-- ============================================================
--  ATUALIZACAO 59 — os pedidos da supervisora (18/11)
--
--  Uma atualizacao so para o dia, uma parte por assunto. Tudo aqui pode
--  rodar de novo sem quebrar nem duplicar nada: se uma parte nova entrar
--  depois, basta rodar o arquivo inteiro outra vez.
--
--  1. CLIENTE: onde o colaborador fica e o faturamento
--     cliente.hospedagem — "Onde o colaborador fica?": 'hotel' ou
--     'airbnb'. NULL = sem resposta (a unidade sem resposta propria vale
--     a do cliente, como o "Oferece alimentacao?").
--     cliente.fatura_apos_documento — "Fatura so depois do documento
--     digital liberado?": true, false ou NULL (sem resposta).
--
--  2. CANCELAR A OBRA
--     obra.cancelamento_motivo — por que a obra foi cancelada (a chave da
--     lista MOTIVOS_CANCELAMENTO, em src/domain/obras.js). A obra
--     cancelada fecha como a concluida (concluida_em, concluida_por e
--     conclusao_obs, que guarda a observacao); o motivo preenchido e o
--     que a marca como cancelada. NULL = nao foi cancelada.
--     A permissao nova `cancelar_obras` vai, uma vez so, para os setores
--     que criam obra (`criar_obras`: o Comercial e o acesso total).
--
--  3. OS AVISOS DO PRAZO DA EXECUCAO (no Teams)
--     Prolongar a execucao (o ultimo dia passa do prazo que ja existia)
--     pede o motivo. Sao duas chaves de AVISO, nao de acesso: vale so o
--     setor que tem a chave marcada (o acesso total nao entra sozinho).
--       aviso_prazo_obra    obra prolongada, encurtada ou com a execucao
--                           concluida antes do prazo: a hospedagem. Vai,
--                           uma vez so, para quem revisa as despesas de
--                           todos (`revisar_despesa_geral`: o Administrativo)
--       aviso_aditivo_obra  obra prolongada, com o motivo: o aditivo. Vai,
--                           uma vez so, para quem edita negocio
--                           (`editar_negocios`: o Comercial)
--     As duas dependem de `ver_obras`: so entram no setor que ve Obras.
--     Nao e coluna nova: e so a lista de permissoes do setor.
--
--  SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon.
-- ============================================================

BEGIN;

-- ---------------- 1. CLIENTE ----------------

ALTER TABLE public.cliente ADD COLUMN IF NOT EXISTS hospedagem text;
ALTER TABLE public.cliente DROP CONSTRAINT IF EXISTS cliente_hospedagem_ok;
ALTER TABLE public.cliente ADD CONSTRAINT cliente_hospedagem_ok
    CHECK (hospedagem IS NULL OR hospedagem IN ('hotel', 'airbnb'));

COMMENT ON COLUMN public.cliente.hospedagem IS
    'Onde o colaborador fica na obra deste cliente: hotel ou airbnb. NULL = sem resposta (a unidade vale a do cliente).';

ALTER TABLE public.cliente ADD COLUMN IF NOT EXISTS fatura_apos_documento boolean;

COMMENT ON COLUMN public.cliente.fatura_apos_documento IS
    'O cliente so fatura depois do documento digital liberado? NULL = sem resposta (a unidade vale a do cliente).';

-- ---------------- 2. CANCELAR A OBRA ----------------

ALTER TABLE public.obra ADD COLUMN IF NOT EXISTS cancelamento_motivo text;
ALTER TABLE public.obra DROP CONSTRAINT IF EXISTS obra_cancelamento_ok;
ALTER TABLE public.obra ADD CONSTRAINT obra_cancelamento_ok
    CHECK (cancelamento_motivo IS NULL OR (concluida_em IS NOT NULL AND char_length(cancelamento_motivo) <= 40));

COMMENT ON COLUMN public.obra.cancelamento_motivo IS
    'Por que a obra foi cancelada (chave de MOTIVOS_CANCELAMENTO). Preenchido = cancelada; ela fecha como a concluida.';

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['cancelar_obras']
 WHERE NOT ('cancelar_obras' = ANY (permissoes))
   AND NOT coalesce(painel_tv, false)
   AND (acesso_total OR 'criar_obras' = ANY (permissoes))
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_59_cancelar');

INSERT INTO public.configuracao (chave, valor)
VALUES ('marco_atualizacao_59_cancelar', 'cancelar_obras distribuida a quem cria obra')
ON CONFLICT (chave) DO NOTHING;

-- ---------------- 3. OS AVISOS DO PRAZO DA EXECUCAO ----------------

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['aviso_prazo_obra']
 WHERE NOT ('aviso_prazo_obra' = ANY (permissoes))
   AND NOT coalesce(painel_tv, false) AND NOT acesso_total
   AND 'ver_obras' = ANY (permissoes) AND 'revisar_despesa_geral' = ANY (permissoes)
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_59_avisos');

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['aviso_aditivo_obra']
 WHERE NOT ('aviso_aditivo_obra' = ANY (permissoes))
   AND NOT coalesce(painel_tv, false) AND NOT acesso_total
   AND 'ver_obras' = ANY (permissoes) AND 'editar_negocios' = ANY (permissoes)
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_59_avisos');

INSERT INTO public.configuracao (chave, valor)
VALUES ('marco_atualizacao_59_avisos', 'aviso_prazo_obra e aviso_aditivo_obra distribuidas')
ON CONFLICT (chave) DO NOTHING;

COMMIT;

-- ============================================================
--  CONFERENCIA — deve devolver "t" em todas as linhas
-- ============================================================

SELECT 'cliente.hospedagem' AS item,
       EXISTS (SELECT 1 FROM information_schema.columns
                WHERE table_name = 'cliente' AND column_name = 'hospedagem') AS ok
UNION ALL SELECT 'cliente.fatura_apos_documento',
       EXISTS (SELECT 1 FROM information_schema.columns
                WHERE table_name = 'cliente' AND column_name = 'fatura_apos_documento')
UNION ALL SELECT 'obra.cancelamento_motivo',
       EXISTS (SELECT 1 FROM information_schema.columns
                WHERE table_name = 'obra' AND column_name = 'cancelamento_motivo')
UNION ALL SELECT 'setores com cancelar_obras: ' || count(*), count(*) > 0
  FROM public.cargo WHERE 'cancelar_obras' = ANY (permissoes)
UNION ALL SELECT 'avisos do prazo distribuidos',
       EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_59_avisos');
