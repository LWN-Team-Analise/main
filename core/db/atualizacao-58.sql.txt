-- ============================================================
--  ATUALIZACAO 58 — o Faturar espera o Administrativo, os tipos dos
--  Turnos e a Area da qualidade
--
--  1. Os checks do faturamento (5a etapa) viram checks do SISTEMA:
--       "Liberado para faturamento" (Administrativo)  -> liberado_faturamento
--       "Faturar após liberação do administrativo"     -> faturar, e passa a
--                                                        se chamar so "Faturar"
--     O Comercial so marca "Faturar" depois que o Administrativo marca a
--     liberacao, e a liberacao nao se desmarca com o "Faturar" marcado
--     (src/domain/obras.js, server/routes/dados.js). Vale para todas as
--     versoes do check e para as copias das obras com checks proprios.
--     Banco sem esses checks fica como esta.
--
--  2. turno.tipo: o turno de trabalho ('turno', o de sempre) ou uma
--     ausencia — 'ferias', 'atestado', 'folga' (folga solicitada) ou
--     'nao_operacional' (dia nao operacional). A cor do turno continua
--     em turno.cor: um dos ids das amostras ou uma cor propria '#rrggbb'
--     (nao muda nada no banco: a coluna ja e texto livre).
--
--  3. A Area da qualidade:
--       qualidade_documento         o acompanhamento dos documentos: cada
--                                   documento anda pelas etapas do funil
--                                   dele (protocolos, relatorios, gerais),
--                                   como os negocios do Comercial;
--       qualidade_documento_evento  o historico (criado, etapa, revisao,
--                                   concluido, comentario...);
--       correio_solicitacao         os pedidos de envio pelo correio.
--
--  4. As permissoes (src/domain/permissoes.js), dadas uma vez so (a marca
--     'marco_atualizacao_58'):
--       ver_documentos, editar_documentos  a Qualidade (gq) e o acesso total;
--       ver_correio                        a Qualidade, quem atende e o
--                                          acesso total;
--       gerenciar_correio                  o acesso total e quem ja revisa as
--                                          despesas de todos (o Admin e o
--                                          Administrativo, que geram a
--                                          etiqueta da pasta de correio).
--     Quem cuida dos setores muda depois, na tela de cada um.
--
--  So acrescenta. SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon. Rodar de novo nao
--  faz nada.
-- ============================================================

BEGIN;

-- ---------- 1. os checks do faturamento ----------

ALTER TABLE public.etapa_check DROP CONSTRAINT IF EXISTS etapa_check_tipo_valido;
ALTER TABLE public.etapa_check ADD CONSTRAINT etapa_check_tipo_valido
    CHECK (tipo IN ('comum', 'planejamento_ensaios', 'execucao_ensaios', 'material_gases',
                    'prazo_execucao', 'reserva_carro', 'retirada_carro', 'enviar_os',
                    'separar_os', 'retirada_os', 'devolucao_os',
                    'aprovacao_protocolos', 'analise_faturamento', 'tipo_ensaio', 'focal_point',
                    'liberado_faturamento', 'faturar'));

COMMENT ON COLUMN public.etapa_check.tipo IS
    'comum, planejamento_ensaios, execucao_ensaios, material_gases, prazo_execucao, reserva_carro, retirada_carro, enviar_os, separar_os, retirada_os, devolucao_os, aprovacao_protocolos, analise_faturamento, tipo_ensaio, focal_point, liberado_faturamento ou faturar';

-- o do Administrativo: todas as versoes dele e as copias das obras
UPDATE public.etapa_check k
   SET tipo = 'liberado_faturamento'
 WHERE k.tipo = 'comum'
   AND lower(btrim(k.titulo)) = 'liberado para faturamento';

-- o do Comercial: vira "Faturar"
UPDATE public.etapa_check k
   SET tipo = 'faturar',
       titulo = 'Faturar'
 WHERE k.tipo = 'comum'
   AND lower(btrim(k.titulo)) LIKE 'faturar ap_s libera__o do administrativo%';

-- ---------- 2. o tipo do turno ----------

ALTER TABLE public.turno ADD COLUMN IF NOT EXISTS tipo text NOT NULL DEFAULT 'turno';
ALTER TABLE public.turno DROP CONSTRAINT IF EXISTS turno_tipo_ok;
ALTER TABLE public.turno ADD CONSTRAINT turno_tipo_ok
    CHECK (tipo IN ('turno', 'ferias', 'atestado', 'folga', 'nao_operacional'));

COMMENT ON COLUMN public.turno.tipo IS
    'turno (de trabalho), ferias, atestado, folga (folga solicitada) ou nao_operacional (dia nao operacional; sem pessoa vale para o cronograma inteiro).';
COMMENT ON COLUMN public.turno.cor IS
    'O id de uma das amostras (src/domain/turnos.js, CORES) ou uma cor propria #rrggbb.';

-- ---------- 3. a Area da qualidade ----------

CREATE TABLE IF NOT EXISTS public.qualidade_documento (
    id              bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    funil           text        NOT NULL,
    etapa           text        NOT NULL,
    titulo          text        NOT NULL,
    tipo            text        NOT NULL DEFAULT 'outro',
    codigo          text,
    revisao         integer     NOT NULL DEFAULT 0,
    obra_id         bigint      REFERENCES public.obra (id) ON DELETE SET NULL,
    cliente_id      bigint      REFERENCES public.cliente (id) ON DELETE SET NULL,
    responsavel_id  bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    prazo           date,
    prioridade      text        NOT NULL DEFAULT 'normal',
    observacao      text,
    status          text        NOT NULL DEFAULT 'aberto',
    -- desde quando esta na etapa de agora (o "ha N dias" do cartao)
    etapa_desde     timestamptz NOT NULL DEFAULT now(),
    criado_por      bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    criado_em       timestamptz NOT NULL DEFAULT now(),
    atualizado_em   timestamptz NOT NULL DEFAULT now(),
    concluido_em    timestamptz,

    CONSTRAINT qualidade_documento_funil_ok CHECK (funil IN ('protocolos', 'relatorios', 'gerais')),
    CONSTRAINT qualidade_documento_titulo_ok CHECK (char_length(btrim(titulo)) BETWEEN 1 AND 200),
    CONSTRAINT qualidade_documento_tipo_ok
        CHECK (tipo IN ('protocolo', 'relatorio', 'certificado', 'laudo', 'procedimento', 'outro')),
    CONSTRAINT qualidade_documento_codigo_ok CHECK (codigo IS NULL OR char_length(codigo) <= 40),
    CONSTRAINT qualidade_documento_revisao_ok CHECK (revisao BETWEEN 0 AND 99),
    CONSTRAINT qualidade_documento_prioridade_ok CHECK (prioridade IN ('baixa', 'normal', 'alta', 'urgente')),
    CONSTRAINT qualidade_documento_observacao_ok CHECK (observacao IS NULL OR char_length(observacao) <= 2000),
    CONSTRAINT qualidade_documento_status_ok CHECK (status IN ('aberto', 'concluido', 'cancelado'))
);

CREATE INDEX IF NOT EXISTS qualidade_documento_abertos ON public.qualidade_documento (funil, etapa) WHERE status = 'aberto';
CREATE INDEX IF NOT EXISTS qualidade_documento_da_obra ON public.qualidade_documento (obra_id) WHERE obra_id IS NOT NULL;

COMMENT ON TABLE public.qualidade_documento IS
    'Acompanhamento dos documentos da Qualidade: cada documento anda pelas etapas do funil dele (src/domain/qualidade.js).';

CREATE TABLE IF NOT EXISTS public.qualidade_documento_evento (
    id            bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    documento_id  bigint      NOT NULL REFERENCES public.qualidade_documento (id) ON DELETE CASCADE,
    usuario_id    bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    tipo          text        NOT NULL,
    de            text,
    para          text,
    texto         text,
    criado_em     timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT qualidade_documento_evento_tipo_ok
        CHECK (tipo IN ('criado', 'editado', 'etapa', 'revisao', 'concluido', 'cancelado', 'reaberto', 'comentario')),
    CONSTRAINT qualidade_documento_evento_texto_ok CHECK (texto IS NULL OR char_length(texto) <= 1000)
);

CREATE INDEX IF NOT EXISTS qualidade_documento_evento_do_documento
    ON public.qualidade_documento_evento (documento_id, criado_em DESC);

COMMENT ON TABLE public.qualidade_documento_evento IS
    'O historico de cada documento da Qualidade: criado, editado, etapa (de/para), revisao, concluido, cancelado, reaberto e comentarios.';

CREATE TABLE IF NOT EXISTS public.correio_solicitacao (
    id              bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    solicitante_id  bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    obra_id         bigint      REFERENCES public.obra (id) ON DELETE SET NULL,
    cliente_id      bigint      REFERENCES public.cliente (id) ON DELETE SET NULL,
    conteudo        text        NOT NULL,
    destinatario    text        NOT NULL,
    contato         text,
    telefone        text,
    cep             text,
    logradouro      text        NOT NULL,
    numero          text,
    complemento     text,
    bairro          text,
    cidade          text        NOT NULL,
    uf              text,
    servico         text        NOT NULL DEFAULT 'sedex',
    -- ate quando precisa sair / chegar
    prazo           date,
    observacao      text,
    situacao        text        NOT NULL DEFAULT 'solicitada',
    situacao_desde  timestamptz NOT NULL DEFAULT now(),
    rastreio        text,
    custo           numeric(10, 2),
    postado_em      date,
    entregue_em     date,
    -- o recado de quem atende (o motivo, no cancelamento)
    resposta        text,
    atendido_por    bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    criado_em       timestamptz NOT NULL DEFAULT now(),
    atualizado_em   timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT correio_solicitacao_conteudo_ok CHECK (char_length(btrim(conteudo)) BETWEEN 1 AND 300),
    CONSTRAINT correio_solicitacao_destinatario_ok CHECK (char_length(btrim(destinatario)) BETWEEN 1 AND 150),
    CONSTRAINT correio_solicitacao_cep_ok CHECK (cep IS NULL OR cep ~ '^[0-9]{8}$'),
    CONSTRAINT correio_solicitacao_servico_ok
        CHECK (servico IN ('sedex', 'sedex10', 'pac', 'carta_registrada', 'motoboy', 'outro')),
    CONSTRAINT correio_solicitacao_situacao_ok
        CHECK (situacao IN ('solicitada', 'preparando', 'postada', 'entregue', 'cancelada')),
    CONSTRAINT correio_solicitacao_observacao_ok CHECK (observacao IS NULL OR char_length(observacao) <= 1000),
    CONSTRAINT correio_solicitacao_resposta_ok CHECK (resposta IS NULL OR char_length(resposta) <= 1000),
    CONSTRAINT correio_solicitacao_rastreio_ok CHECK (rastreio IS NULL OR char_length(rastreio) <= 40),
    CONSTRAINT correio_solicitacao_custo_ok CHECK (custo IS NULL OR custo BETWEEN 0 AND 99999)
);

CREATE INDEX IF NOT EXISTS correio_solicitacao_situacao ON public.correio_solicitacao (situacao, situacao_desde);

COMMENT ON TABLE public.correio_solicitacao IS
    'Pedido de envio pelo correio: o que vai, para quem e para onde; quem atende (gerenciar_correio) leva pelas situacoes.';

-- ---------- 4. as permissoes ----------

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['ver_documentos']
 WHERE NOT ('ver_documentos' = ANY (permissoes))
   AND NOT coalesce(painel_tv, false)
   AND (acesso_total OR chave = 'gq')
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_58');

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['editar_documentos']
 WHERE NOT ('editar_documentos' = ANY (permissoes))
   AND NOT coalesce(painel_tv, false)
   AND (acesso_total OR chave = 'gq')
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_58');

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['ver_correio']
 WHERE NOT ('ver_correio' = ANY (permissoes))
   AND NOT coalesce(painel_tv, false)
   AND (acesso_total OR chave = 'gq' OR 'revisar_despesa_geral' = ANY (permissoes))
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_58');

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['gerenciar_correio']
 WHERE NOT ('gerenciar_correio' = ANY (permissoes))
   AND NOT coalesce(painel_tv, false)
   AND (acesso_total OR 'revisar_despesa_geral' = ANY (permissoes))
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_58');

INSERT INTO public.configuracao (chave, valor)
VALUES ('marco_atualizacao_58', 'ver_documentos, editar_documentos, ver_correio e gerenciar_correio distribuidas')
ON CONFLICT (chave) DO NOTHING;

COMMIT;

-- ============================================================
--  CONFERENCIA
-- ============================================================

SELECT 'check liberado_faturamento' AS item,
       EXISTS (SELECT 1 FROM public.etapa_check WHERE tipo = 'liberado_faturamento') AS ok
UNION ALL SELECT 'check faturar (Faturar)',
       EXISTS (SELECT 1 FROM public.etapa_check WHERE tipo = 'faturar' AND titulo = 'Faturar')
UNION ALL SELECT 'turno.tipo',
       EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name = 'turno' AND column_name = 'tipo')
UNION ALL SELECT 'qualidade_documento', to_regclass('public.qualidade_documento') IS NOT NULL
UNION ALL SELECT 'qualidade_documento_evento', to_regclass('public.qualidade_documento_evento') IS NOT NULL
UNION ALL SELECT 'correio_solicitacao', to_regclass('public.correio_solicitacao') IS NOT NULL
UNION ALL SELECT 'setores com ver_documentos: ' || count(*), count(*) > 0
  FROM public.cargo WHERE 'ver_documentos' = ANY (permissoes)
UNION ALL SELECT 'setores com gerenciar_correio: ' || count(*), count(*) > 0
  FROM public.cargo WHERE 'gerenciar_correio' = ANY (permissoes);
