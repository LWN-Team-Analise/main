-- ============================================================
--  ATUALIZACAO 27 — COMERCIAL, PESSOAS DO CLIENTE E A TELA DA TV
--
--  1. cliente_pessoa — as pessoas de contato de cada cliente (nome,
--     cargo, e-mail, telefone). Um cliente tem varias.
--
--  2. cliente.proprietario_id / cliente.proximo_contato — quem
--     cadastrou (ou responde por) a empresa e a data do proximo
--     contato. Os clientes que ja existiam ganham como proprietario
--     quem os cadastrou, lido do historico (atividade 'cliente.criado').
--
--  3. negocio — os negocios da aba Comercial, cada um num dos quatro
--     funis (novos, renovacao, resgate, prospeccao) e numa etapa dele.
--     A lista de etapas mora no codigo (src/domain/comercial.js), como
--     a de permissoes; o banco guarda so a chave.
--     As colunas email_* guardam o ultimo e-mail do contato que ainda
--     espera resposta (o "Responder?" do cartao), quando a API da
--     Microsoft pode ler a caixa de entrada.
--
--  4. negocio_email — os e-mails mandados pelo site, saindo da caixa
--     do proprio usuario.
--
--  5. cargo.painel_tv — o setor da televisao da empresa: so a pagina
--     inicial, alternando Grade e Quadro sozinha, sem login do Outlook.
--     Nasce o setor "Painel TV" com essa marca.
--
--  6. As permissoes novas: ver_historico vai para todo setor que ja
--     via a pagina inicial (o Historico deixou de ser de todos para a
--     TV nao mostrar); exportar_clientes para quem edita cliente; e a
--     aba Comercial (ver, editar, excluir negocio e responder e-mail)
--     para o setor Comercial e para o acesso total. Isso roda UMA vez
--     so (marca 'marco_atualizacao_27' em configuracao): rodar de novo
--     nao devolve uma permissao que alguem tirou depois.
--
--  SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon.
--
--  Rodar uma vez. Rodar de novo nao quebra nada e nao duplica nada.
-- ============================================================

BEGIN;

-- ---- 1. as pessoas de contato do cliente ----
CREATE TABLE IF NOT EXISTS public.cliente_pessoa (
    id          bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    cliente_id  bigint      NOT NULL REFERENCES public.cliente (id) ON DELETE CASCADE,
    nome        text        NOT NULL,
    cargo       text,
    email       text,
    telefone    text,
    ordem       smallint    NOT NULL DEFAULT 0,
    criado_em   timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT cliente_pessoa_nome_tam CHECK (char_length(nome) BETWEEN 1 AND 200)
);

CREATE INDEX IF NOT EXISTS cliente_pessoa_do_cliente ON public.cliente_pessoa (cliente_id, ordem);

COMMENT ON TABLE public.cliente_pessoa IS
    'Pessoas de contato do cliente. O e-mail delas e o destino do "Responder?" dos negocios.';

-- ---- 2. dono e proximo contato do cliente ----
ALTER TABLE public.cliente ADD COLUMN IF NOT EXISTS proprietario_id bigint REFERENCES public.usuario (id) ON DELETE SET NULL;
ALTER TABLE public.cliente ADD COLUMN IF NOT EXISTS proximo_contato date;

COMMENT ON COLUMN public.cliente.proprietario_id IS 'Quem cadastrou (ou responde por) a empresa no sistema.';
COMMENT ON COLUMN public.cliente.proximo_contato IS 'Data do proximo contato com a empresa.';

-- ---- 3. os negocios ----
CREATE TABLE IF NOT EXISTS public.negocio (
    id                  bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    funil               text          NOT NULL,
    etapa               text          NOT NULL,
    titulo              text          NOT NULL,
    cliente_id          bigint        REFERENCES public.cliente (id) ON DELETE SET NULL,
    -- o nome da empresa quando ela ainda nao e cliente cadastrado (prospeccao)
    organizacao         text,
    pessoa_id           bigint        REFERENCES public.cliente_pessoa (id) ON DELETE SET NULL,
    contato_nome        text,
    contato_email       text,
    contato_telefone    text,
    valor               numeric(14,2) NOT NULL DEFAULT 0,
    responsavel_id      bigint        REFERENCES public.usuario (id) ON DELETE SET NULL,
    etiqueta            text,
    proxima_atividade   date,
    atividade_descricao text,
    previsao_fechamento date,
    status              text          NOT NULL DEFAULT 'aberto',
    motivo_perda        text,
    fechado_em          timestamptz,
    observacao          text,
    ordem               integer       NOT NULL DEFAULT 0,
    email_aguardando    boolean       NOT NULL DEFAULT false,
    email_mensagem_id   text,
    email_assunto       text,
    email_recebido_em   timestamptz,
    email_conferido_em  timestamptz,
    criado_por          bigint        REFERENCES public.usuario (id) ON DELETE SET NULL,
    criado_em           timestamptz   NOT NULL DEFAULT now(),
    atualizado_em       timestamptz   NOT NULL DEFAULT now(),

    CONSTRAINT negocio_funil_valido  CHECK (funil IN ('novos', 'renovacao', 'resgate', 'prospeccao')),
    CONSTRAINT negocio_status_valido CHECK (status IN ('aberto', 'ganho', 'perdido')),
    CONSTRAINT negocio_valor_positivo CHECK (valor >= 0),
    CONSTRAINT negocio_titulo_tam    CHECK (char_length(titulo) BETWEEN 1 AND 200)
);

CREATE INDEX IF NOT EXISTS negocio_do_funil   ON public.negocio (funil, status, etapa, ordem);
CREATE INDEX IF NOT EXISTS negocio_do_cliente ON public.negocio (cliente_id);

COMMENT ON TABLE public.negocio IS
    'Negocios da aba Comercial. funil/etapa: chaves de src/domain/comercial.js. status: aberto, ganho ou perdido.';

-- ---- 4. os e-mails mandados pelo site ----
CREATE TABLE IF NOT EXISTS public.negocio_email (
    id            bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    negocio_id    bigint      NOT NULL REFERENCES public.negocio (id) ON DELETE CASCADE,
    usuario_id    bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    de            text        NOT NULL,
    para          text        NOT NULL,
    assunto       text        NOT NULL,
    corpo         text        NOT NULL,
    -- o id (Microsoft Graph) da mensagem respondida, quando foi resposta na conversa
    em_resposta_a text,
    enviado_em    timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS negocio_email_do_negocio ON public.negocio_email (negocio_id, enviado_em DESC);

COMMENT ON TABLE public.negocio_email IS
    'E-mails mandados ao contato do negocio pelo site, saindo da caixa do proprio usuario.';

-- ---- 5. o setor da televisao ----
ALTER TABLE public.cargo ADD COLUMN IF NOT EXISTS painel_tv boolean NOT NULL DEFAULT false;

COMMENT ON COLUMN public.cargo.painel_tv IS
    'true = tela da TV: so a pagina inicial, alternando Grade e Quadro sozinha, e sem login do Outlook.';

-- ---- 6. as permissoes novas, uma vez so ----
UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['ver_historico']
 WHERE 'ver_inicio' = ANY (permissoes)
   AND NOT ('ver_historico' = ANY (permissoes))
   AND NOT painel_tv
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_27');

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['exportar_clientes']
 WHERE 'editar_clientes' = ANY (permissoes)
   AND NOT ('exportar_clientes' = ANY (permissoes))
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_27');

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY(
           SELECT p FROM unnest(ARRAY['ver_comercial', 'editar_negocios', 'excluir_negocios', 'responder_emails']) AS p
            WHERE NOT (p = ANY (permissoes))
       )
 WHERE (acesso_total OR chave = 'comercial')
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_27');

-- o acesso total ja passa por todas; elas entram na lista so para a tela
-- de Setores mostrar as caixas marcadas
UPDATE public.cargo
   SET permissoes = permissoes || ARRAY(
           SELECT p FROM unnest(ARRAY['ver_historico', 'exportar_clientes']) AS p
            WHERE NOT (p = ANY (permissoes))
       )
 WHERE acesso_total
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_27');

-- quem cadastrou cada cliente que ja existia (o historico guarda)
UPDATE public.cliente c
   SET proprietario_id = a.usuario_id
  FROM (SELECT DISTINCT ON (entidade_id) entidade_id, usuario_id
          FROM public.atividade
         WHERE acao = 'cliente.criado'
         ORDER BY entidade_id, criado_em) a
 WHERE c.proprietario_id IS NULL
   AND a.entidade_id = c.id::text
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_27');

INSERT INTO public.configuracao (chave, valor)
VALUES ('marco_atualizacao_27', 'permissoes da atualizacao 27 distribuidas')
ON CONFLICT (chave) DO NOTHING;

-- o setor da TV: so a pagina inicial (sem painel de graficos e sem historico)
INSERT INTO public.cargo (chave, nome, curto, cor, acesso_total, fixo, ordem, permissoes, painel_tv)
VALUES ('painel_tv', 'Painel TV', 'tv', '#64748b', false, false, 90, ARRAY['ver_inicio'], true)
ON CONFLICT (chave) DO NOTHING;

COMMIT;

-- ============================================================
--  CONFERENCIA — deve devolver "t" em todas as linhas
-- ============================================================

SELECT 'cliente_pessoa' AS item, to_regclass('public.cliente_pessoa') IS NOT NULL AS ok
UNION ALL SELECT 'negocio', to_regclass('public.negocio') IS NOT NULL
UNION ALL SELECT 'negocio_email', to_regclass('public.negocio_email') IS NOT NULL
UNION ALL SELECT 'cliente.proprietario_id',
                 EXISTS (SELECT 1 FROM information_schema.columns
                          WHERE table_name = 'cliente' AND column_name = 'proprietario_id')
UNION ALL SELECT 'cargo.painel_tv',
                 EXISTS (SELECT 1 FROM information_schema.columns
                          WHERE table_name = 'cargo' AND column_name = 'painel_tv')
UNION ALL SELECT 'setor Painel TV', EXISTS (SELECT 1 FROM public.cargo WHERE chave = 'painel_tv' AND painel_tv)
UNION ALL SELECT 'ver_comercial no acesso total',
                 NOT EXISTS (SELECT 1 FROM public.cargo WHERE acesso_total AND NOT ('ver_comercial' = ANY (permissoes)));
