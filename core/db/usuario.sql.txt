-- =====================================================================

--  As aspas sao obrigatorias: o nome tem letras maiusculas.

CREATE DATABASE "TrajetoClientes";


-- =====================================================================
--  PARTE 2 - reconecte no banco "TrajetoClientes" e rode daqui em diante
-- =====================================================================

CREATE TABLE usuario (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name            TEXT        NOT NULL,
    email           TEXT        NOT NULL,
    cpf             CHAR(11)    NOT NULL UNIQUE,
    data_nascimento DATE        NOT NULL,
    telefone        TEXT,
    cargo           TEXT        NOT NULL,
    permissoes      TEXT[]      NOT NULL DEFAULT '{}',
    -- avaliacao de 0 a 10 (aceita meio ponto: 8.5). NULL = ainda nao avaliado
    avaliacao       NUMERIC(3,1),
    -- true = tem login do Outlook | false = nao tem | NULL = a confirmar
    outlook         BOOLEAN,

    -- ---- colunas de apoio (nao estavam na sua lista, mas o login precisa) ----
    senha_hash      TEXT        NOT NULL,
    ativo           BOOLEAN     NOT NULL DEFAULT true,
    protegido       BOOLEAN     NOT NULL DEFAULT false,
    ultimo_acesso   TIMESTAMPTZ,
    criado_em       TIMESTAMPTZ NOT NULL DEFAULT now(),
    atualizado_em   TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT usuario_cpf_digitos     CHECK (cpf ~ '^[0-9]{11}$'),
    CONSTRAINT usuario_email_forma     CHECK (email ~* '^[^@[:space:]]+@[^@[:space:]]+[.][a-z]{2,}$'),
    CONSTRAINT usuario_avaliacao_faixa CHECK (avaliacao IS NULL OR (avaliacao >= 0 AND avaliacao <= 10))
);

COMMENT ON COLUMN usuario.cpf       IS 'Somente digitos (11). A mascara e aplicada na interface.';
COMMENT ON COLUMN usuario.avaliacao IS 'Nota de 0 a 10. NULL = ainda nao avaliado.';
COMMENT ON COLUMN usuario.outlook   IS 'Possui login do Outlook. NULL = a confirmar.';
COMMENT ON COLUMN usuario.protegido IS 'Registro travado: so muda com app.desbloqueio = on.';

-- e-mail unico ignorando maiusculas/minusculas; serve tambem a busca do login
CREATE UNIQUE INDEX usuario_email_uk ON usuario (lower(email));


-- ---------------------------------------------------------------------
--  Carimbo automatico de atualizacao
-- ---------------------------------------------------------------------

CREATE OR REPLACE FUNCTION usuario_toca_atualizado() RETURNS trigger AS $fn$
BEGIN
    NEW.atualizado_em := now();
    RETURN NEW;
END;
$fn$ LANGUAGE plpgsql;

CREATE TRIGGER trg_a_usuario_toca
    BEFORE UPDATE ON usuario
    FOR EACH ROW
    EXECUTE FUNCTION usuario_toca_atualizado();


-- ---------------------------------------------------------------------
--  Protecao do registro (usada no usuario id = 1)
--
--  Bloqueia UPDATE e DELETE de qualquer linha com protegido = true.
--  A unica excecao automatica e o carimbo de login (ultimo_acesso), para
--  o sistema conseguir registrar o acesso sem destravar nada.
--
--  Para editar/excluir de proposito, destrave na propria transacao:
--      BEGIN;
--      SET LOCAL app.desbloqueio = 'on';
--      ... comando ...
--      COMMIT;
-- ---------------------------------------------------------------------

CREATE OR REPLACE FUNCTION usuario_protege() RETURNS trigger AS $fn$
BEGIN
    -- destravado de proposito nesta transacao
    IF coalesce(current_setting('app.desbloqueio', true), '') = 'on' THEN
        RETURN CASE WHEN TG_OP = 'DELETE' THEN OLD ELSE NEW END;
    END IF;

    IF TG_OP = 'DELETE' THEN
        RAISE EXCEPTION
            'Usuario % e protegido: exclusao exige SET LOCAL app.desbloqueio = on.', OLD.id
            USING ERRCODE = 'insufficient_privilege';
    END IF;

    -- UPDATE: libera apenas se nenhum campo de conteudo mudou
    -- (ou seja, mexeu so em ultimo_acesso / atualizado_em)
    IF ROW(NEW.id, NEW.name, NEW.email, NEW.cpf, NEW.data_nascimento, NEW.telefone,
           NEW.cargo, NEW.permissoes, NEW.avaliacao, NEW.outlook, NEW.senha_hash,
           NEW.ativo, NEW.protegido)
       IS NOT DISTINCT FROM
       ROW(OLD.id, OLD.name, OLD.email, OLD.cpf, OLD.data_nascimento, OLD.telefone,
           OLD.cargo, OLD.permissoes, OLD.avaliacao, OLD.outlook, OLD.senha_hash,
           OLD.ativo, OLD.protegido)
    THEN
        RETURN NEW;
    END IF;

    RAISE EXCEPTION
        'Usuario % e protegido: edicao exige SET LOCAL app.desbloqueio = on.', OLD.id
        USING ERRCODE = 'insufficient_privilege';
END;
$fn$ LANGUAGE plpgsql;

CREATE TRIGGER trg_b_usuario_protege
    BEFORE UPDATE OR DELETE ON usuario
    FOR EACH ROW
    WHEN (OLD.protegido)
    EXECUTE FUNCTION usuario_protege();
