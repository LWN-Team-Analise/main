-- ============================================================
--  ATUALIZACAO 22 — FROTAS
--
--  A aba Frotas: o cadastro dos veiculos da empresa e o registro de
--  cada SAIDA e DEVOLUCAO, com o condutor, o KM, a obra de destino,
--  as respostas da devolucao e as cinco fotos do carro.
--
--  Tres tabelas:
--
--    veiculo           o carro. `km_atual` e o KM do ultimo registro
--                      dele — o piso da proxima saida: o KM nunca
--                      anda para tras.
--    veiculo_registro  uma saida, uma devolucao ou uma troca de
--                      condutor (o carro ja estava fora e passou
--                      para outra pessoa).
--    veiculo_foto      as fotos de um registro: frente, traseira,
--                      os dois lados e o painel com o KM.
--
--  O veiculo esta EM USO quando o ultimo registro dele nao e uma
--  devolucao. Isso nao e gravado em coluna nenhuma: sai da propria
--  lista de registros, entao nao ha como ficar dessincronizado.
--
--  Os nomes (de quem registrou, do condutor, da obra) vao junto com
--  os ids, como foto do que era na hora: o historico continua
--  legivel se a pessoa for desativada ou a obra apagada. E e assim
--  que entram os registros da planilha antiga do Forms, onde o
--  condutor era um nome digitado e o destino um texto livre
--  (server/scripts/importar-frota.js).
--
--  Veiculo e registro nunca sao apagados em cascata: registro de
--  frota e rastreabilidade. Carro vendido e DESATIVADO (ativo).
--
--  SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon.
--
--  Rodar uma vez. Rodar de novo nao quebra nada e nao apaga nada.
-- ============================================================

BEGIN;

-- ---- o veiculo ----
CREATE TABLE IF NOT EXISTS public.veiculo (
    id             bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome           text NOT NULL,
    marca          text,
    modelo         text,
    placa          text,
    ano            integer,
    cor            text,
    km_atual       integer,
    observacao     text,
    ativo          boolean NOT NULL DEFAULT true,
    criado_em      timestamp with time zone NOT NULL DEFAULT now(),
    atualizado_em  timestamp with time zone NOT NULL DEFAULT now(),

    CONSTRAINT veiculo_nome_tam CHECK (char_length(btrim(nome)) BETWEEN 1 AND 60),
    CONSTRAINT veiculo_km_ok CHECK (km_atual IS NULL OR km_atual BETWEEN 0 AND 9999999),
    CONSTRAINT veiculo_ano_ok CHECK (ano IS NULL OR ano BETWEEN 1950 AND 2100),
    CONSTRAINT veiculo_observacao_tam CHECK (observacao IS NULL OR char_length(observacao) <= 1000)
);

-- "Spin 01" e "spin 01 " sao o mesmo carro; placa tambem nao repete
CREATE UNIQUE INDEX IF NOT EXISTS veiculo_nome_un ON public.veiculo (lower(btrim(nome)));
CREATE UNIQUE INDEX IF NOT EXISTS veiculo_placa_un ON public.veiculo (upper(placa)) WHERE placa IS NOT NULL;

DROP TRIGGER IF EXISTS veiculo_atualizado_tg ON public.veiculo;
CREATE TRIGGER veiculo_atualizado_tg
    BEFORE UPDATE ON public.veiculo
    FOR EACH ROW EXECUTE FUNCTION toca_atualizado_em();

-- ---- o registro: saida, devolucao ou troca ----
CREATE TABLE IF NOT EXISTS public.veiculo_registro (
    id              bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    veiculo_id      bigint NOT NULL REFERENCES public.veiculo(id) ON DELETE RESTRICT,
    tipo            text NOT NULL,
    -- quem registrou (no sistema, o dono do token)
    usuario_id      bigint REFERENCES public.usuario(id) ON DELETE SET NULL,
    usuario_nome    text,
    condutor_id     bigint REFERENCES public.usuario(id) ON DELETE SET NULL,
    condutor_nome   text NOT NULL,
    km              integer,
    data            date NOT NULL,
    hora            time,
    -- so na saida e na troca
    obra_id         bigint REFERENCES public.obra(id) ON DELETE SET NULL,
    destino         text,
    observacao      text,
    -- so na devolucao
    pneus           boolean,
    oleo            boolean,
    agua            boolean,
    abastecido      boolean,
    sem_parar       boolean,
    posto           text,
    limpo_interno   boolean,
    limpo_externo   boolean,
    farois          boolean,
    -- os registros da planilha antiga: as fotos ficaram no SharePoint
    fotos_externas  text[],
    planilha_id     integer,
    criado_em       timestamp with time zone NOT NULL DEFAULT now(),

    CONSTRAINT veiculo_registro_tipo_ok CHECK (tipo IN ('saida', 'devolucao', 'troca')),
    CONSTRAINT veiculo_registro_km_ok CHECK (km IS NULL OR km BETWEEN 0 AND 9999999),
    CONSTRAINT veiculo_registro_condutor_tam CHECK (char_length(btrim(condutor_nome)) BETWEEN 1 AND 200),
    CONSTRAINT veiculo_registro_destino_tam CHECK (destino IS NULL OR char_length(destino) <= 500),
    CONSTRAINT veiculo_registro_observacao_tam CHECK (observacao IS NULL OR char_length(observacao) <= 2000),
    CONSTRAINT veiculo_registro_posto_tam CHECK (posto IS NULL OR char_length(posto) <= 300)
);

-- linha da planilha importada uma vez so
CREATE UNIQUE INDEX IF NOT EXISTS veiculo_registro_planilha_un
    ON public.veiculo_registro (planilha_id) WHERE planilha_id IS NOT NULL;

-- o ultimo registro de cada carro e a leitura de toda tela
CREATE INDEX IF NOT EXISTS veiculo_registro_veiculo_idx
    ON public.veiculo_registro (veiculo_id, data DESC, hora DESC, id DESC);

CREATE INDEX IF NOT EXISTS veiculo_registro_data_idx
    ON public.veiculo_registro (data DESC, id DESC);

-- sem estes, apagar uma obra ou uma pessoa varreria a tabela no SET NULL
CREATE INDEX IF NOT EXISTS veiculo_registro_obra_idx ON public.veiculo_registro (obra_id);
CREATE INDEX IF NOT EXISTS veiculo_registro_usuario_idx ON public.veiculo_registro (usuario_id);
CREATE INDEX IF NOT EXISTS veiculo_registro_condutor_idx ON public.veiculo_registro (condutor_id);

-- ---- as fotos ----
--
-- Data URL, igual aos outros anexos do sistema. Nunca vem na lista:
-- a foto so e lida quando alguem abre o registro.
CREATE TABLE IF NOT EXISTS public.veiculo_foto (
    id           bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    registro_id  bigint NOT NULL REFERENCES public.veiculo_registro(id) ON DELETE CASCADE,
    posicao      text NOT NULL,
    tipo         text,
    tamanho      integer,
    conteudo     text NOT NULL,

    CONSTRAINT veiculo_foto_posicao_ok CHECK (posicao IN ('frente', 'traseira', 'esquerda', 'direita', 'painel')),
    CONSTRAINT veiculo_foto_data_url CHECK (conteudo LIKE 'data:image/%'),
    CONSTRAINT veiculo_foto_uma_por_posicao UNIQUE (registro_id, posicao)
);

COMMENT ON TABLE public.veiculo IS
    'Os veiculos da frota. km_atual = KM do ultimo registro; a proxima saida nao pode ter KM menor.';
COMMENT ON TABLE public.veiculo_registro IS
    'Saidas, devolucoes e trocas de condutor. Em uso = o ultimo registro do veiculo nao e devolucao.';
COMMENT ON COLUMN public.veiculo_registro.planilha_id IS
    'O Id da resposta no Microsoft Forms, para os registros que vieram da planilha antiga.';
COMMENT ON TABLE public.veiculo_foto IS
    'As cinco fotos de cada saida e devolucao, em data URL. Teto de 600 KB por foto, conferido na API.';

-- ---- as permissoes novas ----
--
-- O setor com acesso total ja passa por elas sem estar na lista;
-- elas entram la so para a tela de Setores mostrar as caixas
-- marcadas. Os outros setores comecam sem: quem administra marca.
UPDATE public.cargo
   SET permissoes = permissoes || ARRAY(
           SELECT p FROM unnest(ARRAY['ver_frotas', 'registrar_frota', 'editar_veiculos']) AS p
            WHERE NOT (p = ANY (permissoes))
       )
 WHERE acesso_total;

COMMIT;

-- ============================================================
--  CONFERENCIA — deve devolver "t" em todas as linhas
-- ============================================================

SELECT 'veiculo' AS item, to_regclass('public.veiculo') IS NOT NULL AS ok
UNION ALL SELECT 'veiculo_registro', to_regclass('public.veiculo_registro') IS NOT NULL
UNION ALL SELECT 'veiculo_foto', to_regclass('public.veiculo_foto') IS NOT NULL
UNION ALL SELECT 'ver_frotas no acesso total',
                 NOT EXISTS (SELECT 1 FROM public.cargo WHERE acesso_total AND NOT ('ver_frotas' = ANY (permissoes)));
