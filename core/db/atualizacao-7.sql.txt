-- ============================================================
--  ATUALIZACAO 7 — ETIQUETAS DE CARD
--
--  Ate aqui so a OBRA recebia etiqueta. Esta atualizacao da a
--  mesma coisa ao CARD do roteiro (o bloco de um setor dentro de
--  uma etapa), e de proposito com CATALOGO PROPRIO.
--
--  Por que nao reaproveitar a tabela `etiqueta` que ja existe:
--  as duas etiquetagens respondem a perguntas diferentes — a da
--  obra diz o que aquela obra e ("retrabalho", "garantia"), a do
--  card diz o que aquele pedaco do roteiro e ("depende do
--  cliente", "precisa de ART"). Compartilhar o catalogo faria a
--  lista de sugestoes de uma aparecer na outra, e uma renomeada
--  do lado da obra mudaria a etiqueta do card sem ninguem pedir.
--
--  Rodar uma vez. Tudo e IF NOT EXISTS: rodar de novo nao
--  quebra nada e nao apaga etiqueta nenhuma.
-- ============================================================

BEGIN;

-- ---- o catalogo, so dos cards ----
CREATE TABLE IF NOT EXISTS public.etiqueta_card (
    id        bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome      text NOT NULL,
    cor       text NOT NULL DEFAULT '#6b7280',
    criado_em timestamp with time zone NOT NULL DEFAULT now(),
    CONSTRAINT etiqueta_card_cor_hex CHECK (cor ~* '^#([0-9a-f]{6}|[0-9a-f]{8})$'),
    CONSTRAINT etiqueta_card_nome_tam CHECK (char_length(btrim(nome)) BETWEEN 1 AND 40)
);

-- o nome e unico sem diferenca de maiuscula, igual ao das obras:
-- e ele que faz "Urgente" e "urgente" serem a mesma etiqueta
CREATE UNIQUE INDEX IF NOT EXISTS etiqueta_card_nome_uk
    ON public.etiqueta_card (lower(btrim(nome)));

-- ---- quem usa qual ----
--
-- `card_id` aponta para etapa_card, e o card sobrevive a edicao:
-- renomear ou trocar o cargo dono do card NAO cria linha nova
-- (ver roteiro.js, PATCH /cards/:id), entao a etiqueta continua
-- colada nele. Apagar o card leva as etiquetas dele junto, que e
-- o que ON DELETE CASCADE faz.
CREATE TABLE IF NOT EXISTS public.card_etiqueta (
    card_id     bigint NOT NULL REFERENCES public.etapa_card(id)     ON DELETE CASCADE,
    etiqueta_id bigint NOT NULL REFERENCES public.etiqueta_card(id)  ON DELETE CASCADE,
    posta_em    timestamp with time zone NOT NULL DEFAULT now(),
    PRIMARY KEY (card_id, etiqueta_id)
);

CREATE INDEX IF NOT EXISTS card_etiqueta_card_idx
    ON public.card_etiqueta (card_id);

COMMENT ON TABLE public.etiqueta_card IS
    'Rotulos dos cards do roteiro. Catalogo SEPARADO do de obras (tabela etiqueta), de proposito.';
COMMENT ON TABLE public.card_etiqueta IS
    'Que card usa que etiqueta de card.';

COMMIT;
