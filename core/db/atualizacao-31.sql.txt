-- ============================================================
--  ATUALIZACAO 31 — O CHECK COM HORARIO E A ABA TURNOS
--
--  1. etapa_check.horario — o check com HORARIO (ex.: "Ponto de
--     encontro"). Marcado no editor do card, clicar no check dentro
--     da obra abre as datas da obra, cada uma com o seu horario e as
--     pessoas daquele dia; e ali que o check e marcado.
--
--  2. obra_horario / obra_horario_membro — o horario de cada data
--     (06:30) e quem vai naquele dia. Cada pessoa de cada data vira um
--     turno na aba Turnos.
--
--  3. A ABA TURNOS (a copia dos Turnos do Teams):
--       cronograma          os grupos (Cronograma LWN, Cronograma
--                           Anapolis...), com quem esta em cada um;
--       turno               cada turno: pessoa (ou "turno aberto"),
--                           dias, horario, titulo, cor e, quando vem
--                           de uma obra, a obra e a tarefa;
--       turno_anotacao      a "Anotacao do dia".
--
--  4. As permissoes: `ver_turnos` (a aba) e `editar_turnos` (criar e
--     mudar turnos, anotacoes e cronogramas). Ver a aba entra em todo
--     setor (menos o da TV); editar, so no acesso total — o resto
--     marca quem administra.
--
--  SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon.
--
--  Rodar uma vez. Rodar de novo nao quebra nada e nao duplica nada.
-- ============================================================

BEGIN;

-- ---- 1. o check com horario ----
ALTER TABLE public.etapa_check ADD COLUMN IF NOT EXISTS horario boolean NOT NULL DEFAULT false;

COMMENT ON COLUMN public.etapa_check.horario IS
    'true = check com horario (ponto de encontro): um horario por data da obra, marcado no pop-up dele.';

-- ---- 2. o horario de cada data e quem vai ----
CREATE TABLE IF NOT EXISTS public.obra_horario (
    obra_id        bigint      NOT NULL REFERENCES public.obra (id)        ON DELETE CASCADE,
    check_id       bigint      NOT NULL REFERENCES public.etapa_check (id) ON DELETE CASCADE,
    dia            date        NOT NULL,
    hora           time        NOT NULL,
    atualizado_por bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    atualizado_em  timestamptz NOT NULL DEFAULT now(),
    PRIMARY KEY (obra_id, check_id, dia)
);

CREATE TABLE IF NOT EXISTS public.obra_horario_membro (
    obra_id    bigint NOT NULL,
    check_id   bigint NOT NULL,
    dia        date   NOT NULL,
    usuario_id bigint NOT NULL REFERENCES public.usuario (id) ON DELETE CASCADE,
    PRIMARY KEY (obra_id, check_id, dia, usuario_id),
    FOREIGN KEY (obra_id, check_id, dia)
        REFERENCES public.obra_horario (obra_id, check_id, dia) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE INDEX IF NOT EXISTS obra_horario_dia_idx ON public.obra_horario (dia);

COMMENT ON TABLE public.obra_horario IS
    'O horario (ponto de encontro) de cada data da obra, num check com horario.';
COMMENT ON TABLE public.obra_horario_membro IS
    'Quem vai em cada data do check com horario. Cada linha vira um turno na aba Turnos.';

-- ---- 3. a aba Turnos ----
CREATE TABLE IF NOT EXISTS public.cronograma (
    id        bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome      text        NOT NULL,
    ordem     integer     NOT NULL DEFAULT 0,
    criado_em timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT cronograma_nome_unico UNIQUE (nome)
);

CREATE TABLE IF NOT EXISTS public.cronograma_membro (
    cronograma_id bigint NOT NULL REFERENCES public.cronograma (id) ON DELETE CASCADE,
    usuario_id    bigint NOT NULL REFERENCES public.usuario (id)    ON DELETE CASCADE,
    PRIMARY KEY (cronograma_id, usuario_id)
);

CREATE TABLE IF NOT EXISTS public.turno (
    id            bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    cronograma_id bigint      NOT NULL REFERENCES public.cronograma (id) ON DELETE CASCADE,
    usuario_id    bigint      REFERENCES public.usuario (id) ON DELETE CASCADE,
    data_inicio   date        NOT NULL,
    data_fim      date        NOT NULL,
    hora_inicio   time,
    hora_fim      time,
    titulo        text        NOT NULL,
    subtitulo     text,
    cor           text        NOT NULL DEFAULT 'azul',
    obra_id       bigint      REFERENCES public.obra (id) ON DELETE SET NULL,
    check_id      bigint      REFERENCES public.etapa_check (id) ON DELETE SET NULL,
    criado_por    bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    criado_em     timestamptz NOT NULL DEFAULT now(),
    atualizado_em timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT turno_datas CHECK (data_fim >= data_inicio),
    CONSTRAINT turno_horas CHECK ((hora_inicio IS NULL) = (hora_fim IS NULL))
);

CREATE INDEX IF NOT EXISTS turno_periodo_idx ON public.turno (data_inicio, data_fim);

CREATE TABLE IF NOT EXISTS public.turno_anotacao (
    dia            date        PRIMARY KEY,
    texto          text        NOT NULL,
    atualizado_por bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    atualizado_em  timestamptz NOT NULL DEFAULT now()
);

COMMENT ON TABLE public.cronograma IS 'Os grupos da aba Turnos (Cronograma LWN, Cronograma Anapolis...).';
COMMENT ON TABLE public.turno IS 'Um turno da aba Turnos. usuario_id NULL = turno aberto. Sem horas = dia inteiro.';
COMMENT ON TABLE public.turno_anotacao IS 'A "Anotacao do dia" da aba Turnos.';

-- os dois cronogramas de hoje
INSERT INTO public.cronograma (nome, ordem) VALUES ('Cronograma LWN', 0), ('Cronograma Anápolis', 1)
ON CONFLICT (nome) DO NOTHING;

-- ---- 4. as permissoes e a equipe no Cronograma LWN, uma vez so ----
-- todo mundo ativo (menos a conta da TV) comeca no Cronograma LWN; quem
-- e de Anapolis passa para la pelo proprio cronograma
INSERT INTO public.cronograma_membro (cronograma_id, usuario_id)
SELECT cr.id, u.id
  FROM public.cronograma cr
  JOIN public.usuario u ON coalesce(u.ativo, true)
  LEFT JOIN public.cargo c ON c.id = u.cargo_id
 WHERE cr.nome = 'Cronograma LWN'
   AND NOT coalesce(c.painel_tv, false)
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_31')
ON CONFLICT DO NOTHING;

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['ver_turnos']
 WHERE NOT painel_tv
   AND NOT ('ver_turnos' = ANY (permissoes))
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_31');

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['editar_turnos']
 WHERE acesso_total
   AND NOT ('editar_turnos' = ANY (permissoes))
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_31');

INSERT INTO public.configuracao (chave, valor)
VALUES ('marco_atualizacao_31', 'permissoes e equipe da atualizacao 31 distribuidas')
ON CONFLICT (chave) DO NOTHING;

COMMIT;

-- ============================================================
--  CONFERENCIA — deve devolver "t" em todas as linhas
-- ============================================================

SELECT 'etapa_check.horario' AS item,
       EXISTS (SELECT 1 FROM information_schema.columns
                WHERE table_name = 'etapa_check' AND column_name = 'horario') AS ok
UNION ALL SELECT 'obra_horario', to_regclass('public.obra_horario') IS NOT NULL
UNION ALL SELECT 'obra_horario_membro', to_regclass('public.obra_horario_membro') IS NOT NULL
UNION ALL SELECT 'cronograma', to_regclass('public.cronograma') IS NOT NULL
UNION ALL SELECT 'turno', to_regclass('public.turno') IS NOT NULL
UNION ALL SELECT 'turno_anotacao', to_regclass('public.turno_anotacao') IS NOT NULL
UNION ALL SELECT 'ver_turnos no acesso total',
       NOT EXISTS (SELECT 1 FROM public.cargo WHERE acesso_total AND NOT ('ver_turnos' = ANY (permissoes)));
