-- ============================================================
--  ATUALIZACAO 51 — a aba PONTO
--
--  1. ponto_dia        — um dia da folha de ponto de um funcionario, como
--                        veio no "Calculo Diario" (o .csv do sistema de
--                        ponto, separado por ";"). A chave e o CPF + o dia:
--                        importar o mesmo periodo de novo troca o que
--                        mudou e nao duplica nada.
--  2. ponto_assinatura — a conferencia da folha: o funcionario assina o
--                        periodo e o nome completo dele fica impresso no
--                        lugar de "Assinatura Funcionario". O `resumo` e a
--                        impressao digital das marcacoes do periodo na
--                        hora da assinatura: se uma importacao nova mudar
--                        algum dia, a assinatura fica marcada como
--                        desatualizada e a pessoa assina de novo.
--
--  A folha liga com o usuario pelo CPF (usuario.cpf, so os 11 digitos).
--  Quem ainda nao tem CPF no cadastro nao ve a propria folha ate o CPF ser
--  incluido — a importacao avisa quem ficou sem usuario.
--
--  A permissao nova (`gerenciar_ponto`, embaixo de Usuarios: ver a folha de
--  todos, importar e exportar) nao precisa de coluna: ela vive na lista
--  cargo.permissoes, como as outras. Ninguem ganha sozinho; o acesso total
--  passa por ela, como passa por todas.
--
--  Tambem desta versao, sem SQL: a obra criada em Obras deixou de virar
--  negocio no Comercial. A atualizacao 48 foi aposentada (nao grava mais
--  nada e saiu do `npm run db:atualizar`). Os negocios que ela criou
--  continuam no Comercial.
--
--  So acrescenta. Pode rodar mais de uma vez.
--  SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon.
-- ============================================================

BEGIN;

-- ---- 1. os dias da folha ----
CREATE TABLE IF NOT EXISTS public.ponto_dia (
    cpf                 text        NOT NULL,
    data                date        NOT NULL,
    -- o funcionario como veio no arquivo (vale o da importacao mais nova)
    nome                text        NOT NULL,
    matricula           text,
    pis                 text,
    cnpj                text,
    admissao            date,
    demissao            date,
    filial              text,
    departamento        text,
    cargo               text,
    -- o dia, com os textos do arquivo do jeito que vieram
    dia_semana          text,
    jornada_esperada    text,
    horas_esperadas     text,
    escala              text,
    natureza            text,
    marcacoes           text,
    jornada_considerada text,
    horas_trabalhadas   text,
    eventos             text,
    centro_custo        text,
    grupos              text,
    importado_em        timestamptz NOT NULL DEFAULT now(),
    importado_por       bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,

    PRIMARY KEY (cpf, data),
    CONSTRAINT ponto_dia_cpf_valido CHECK (cpf ~ '^[0-9]{11}$')
);

CREATE INDEX IF NOT EXISTS ponto_dia_da_data ON public.ponto_dia (data);

COMMENT ON TABLE public.ponto_dia IS
    'Folha de ponto: um dia de um funcionario, importado do Calculo Diario (.csv). Liga com usuario.cpf.';

-- ---- 2. as assinaturas ----
CREATE TABLE IF NOT EXISTS public.ponto_assinatura (
    id           bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    usuario_id   bigint      NOT NULL REFERENCES public.usuario (id) ON DELETE CASCADE,
    cpf          text        NOT NULL,
    inicio       date        NOT NULL,
    fim          date        NOT NULL,
    -- o nome completo impresso na folha, como estava na hora de assinar
    nome         text        NOT NULL,
    -- md5 das marcacoes do periodo na hora de assinar (ver server/routes/ponto.js)
    resumo       text        NOT NULL,
    assinado_em  timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT ponto_assinatura_periodo CHECK (fim >= inicio),
    CONSTRAINT ponto_assinatura_cpf_valido CHECK (cpf ~ '^[0-9]{11}$')
);

CREATE INDEX IF NOT EXISTS ponto_assinatura_do_cpf ON public.ponto_assinatura (cpf, inicio, fim);

COMMENT ON TABLE public.ponto_assinatura IS
    'Conferencia da folha de ponto: o proprio funcionario assina um periodo. So ele assina a dele.';

COMMIT;

-- ============================================================
--  CONFERENCIA
-- ============================================================

SELECT 'ponto_dia' AS item, to_regclass('public.ponto_dia') IS NOT NULL AS ok
UNION ALL SELECT 'ponto_assinatura', to_regclass('public.ponto_assinatura') IS NOT NULL;
