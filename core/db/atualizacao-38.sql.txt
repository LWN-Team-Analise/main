-- ---- 2. a abreviacao do cliente e as views ----

ALTER TABLE public.cliente ADD COLUMN IF NOT EXISTS abreviacao text;

COMMENT ON COLUMN public.cliente.abreviacao IS
    'A sigla que o Almoxarife mostra (ABL, BMS...). NULL = o nome.';

CREATE OR REPLACE VIEW almox.clientes AS
SELECT c.id,
       CASE WHEN c.matriz_id IS NULL THEN c.nome
            ELSE concat_ws(' ', m.nome, nullif(btrim(c.unidade), ''))
                 || coalesce(' - ' || nullif(btrim(c.servico), ''), '')
       END AS nome,
       coalesce(nullif(btrim(c.abreviacao), ''),
                CASE WHEN c.matriz_id IS NULL THEN c.nome
                     ELSE concat_ws(' ', m.nome, nullif(btrim(c.unidade), '')) END) AS abreviacao,
       c.cidade,
       CASE WHEN coalesce(c.pais, 'BR') = 'BR' THEN upper(c.estado) END AS uf,
       true AS ativo,
       EXISTS (SELECT 1 FROM public.cliente u WHERE u.matriz_id = c.id) AS tem_unidades
  FROM public.cliente c
  LEFT JOIN public.cliente m ON m.id = c.matriz_id;

COMMENT ON VIEW almox.clientes IS
    'Os clientes do main no formato do Almoxarife (nome, abreviacao, cidade, uf, ativo).';

CREATE OR REPLACE VIEW almox.usuarios AS
SELECT u.id,
       u.name AS nome,
       u.cpf,
       u.email,
       u.telefone,
       coalesce(nullif(btrim(t.nome), ''), nullif(btrim(u.cargo_titulo), ''), c.nome, u.cargo) AS cargo,
       c.nome AS setor,
       coalesce(u.ativo, true) AS ativo,
       u.foto,
       u.criado_em,
       u.atualizado_em,
       to_jsonb(CASE WHEN coalesce(c.acesso_total, false)
                     THEN ARRAY['dashboard', 'solicitacoes', 'gerenciar_os', 'aprovar_todas_os', 'conferencia',
                                'separar_tags', 'devolutiva', 'conferir_devolucao', 'prorrogar_os',
                                'aceitar_prorrogacao', 'aprovar_remanejamento', 'concluidos', 'certificados',
                                'baias', 'instrumentos', 'remanejamento', 'calibracao', 'clientes', 'relatorios',
                                'manutencao', 'logs', 'responsavel_obra']
                     ELSE ARRAY(SELECT substr(p, 7) FROM unnest(coalesce(c.permissoes, '{}'::text[])) AS p
                                 WHERE p LIKE 'almox\_%')
                END) AS permissoes,
       (coalesce(c.acesso_total, false) OR 'almox_responsavel_obra' = ANY (coalesce(c.permissoes, '{}'::text[])))
           AS responsavel_obra
  FROM public.usuario u
  LEFT JOIN public.cargo c        ON c.id = u.cargo_id
  LEFT JOIN public.cargo_titulo t ON t.id = u.cargo_titulo_id
 WHERE NOT coalesce(c.painel_tv, false);

COMMENT ON VIEW almox.usuarios IS
    'Os colaboradores do main no formato do Almoxarife. permissoes = as chaves almox_* do setor, sem o prefixo (acesso total: todas, menos manutencao_somente_leitura).';

-- ---- 4. o check "Enviar OS" ----

ALTER TABLE public.etapa_check DROP CONSTRAINT IF EXISTS etapa_check_tipo_valido;
ALTER TABLE public.etapa_check ADD CONSTRAINT etapa_check_tipo_valido
    CHECK (tipo IN ('comum', 'planejamento_ensaios', 'execucao_ensaios', 'material_gases',
                    'prazo_execucao', 'reserva_carro', 'retirada_carro', 'enviar_os'));

COMMENT ON COLUMN public.etapa_check.tipo IS
    'comum, planejamento_ensaios, execucao_ensaios, material_gases, prazo_execucao, reserva_carro, retirada_carro ou enviar_os. Todos menos comum sao do sistema: nao se excluem.';

-- "Enviar OS": no card da Execucao dos ensaios (Time Tecnico, 3a etapa), no fim
INSERT INTO public.etapa_check (card_id, ordem, titulo, tipo, vigente_de)
SELECT ex.card_id,
       (SELECT coalesce(max(x.ordem), 0) + 1 FROM public.etapa_check x
         WHERE x.card_id = ex.card_id AND x.vigente_ate IS NULL AND x.obra_id IS NULL),
       'Enviar OS', 'enviar_os', now()
  FROM public.etapa_check ex
 WHERE ex.tipo = 'execucao_ensaios'
   AND ex.vigente_ate IS NULL
   AND ex.obra_id IS NULL
   AND NOT EXISTS (SELECT 1 FROM public.etapa_check x
                    WHERE x.tipo = 'enviar_os' AND x.vigente_ate IS NULL AND x.obra_id IS NULL)
 ORDER BY ex.id
 LIMIT 1;

-- ---- 5. as permissoes, distribuidas uma vez ----

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY(
           SELECT p FROM unnest(ARRAY['ver_almoxarife',
               'almox_dashboard', 'almox_solicitacoes', 'almox_gerenciar_os', 'almox_aprovar_todas_os',
               'almox_conferencia', 'almox_separar_tags', 'almox_devolutiva', 'almox_conferir_devolucao',
               'almox_prorrogar_os', 'almox_aceitar_prorrogacao', 'almox_aprovar_remanejamento',
               'almox_concluidos', 'almox_certificados', 'almox_baias', 'almox_instrumentos',
               'almox_remanejamento', 'almox_calibracao', 'almox_clientes', 'almox_relatorios',
               'almox_manutencao', 'almox_logs', 'almox_responsavel_obra']) AS p
            WHERE NOT (p = ANY (permissoes)))
 WHERE (acesso_total OR chave IN ('almoxarife', 'excelencia'))
