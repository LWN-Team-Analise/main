-- ============================================================
--  ATUALIZACAO 52 — a permissao da area ADMINISTRATIVO
--
--  A aba Administrativo (a area dos cards: Reembolso, Ponto, Pontos de
--  encontro, Frotas e Turnos) passou a ter permissao propria, a
--  `ver_administrativo`, no quadro de permissoes do setor (Visualizacao).
--  Quem nao tem a area ve no menu so o Ponto, a propria folha.
--
--  Para ninguem perder o que ja usava, ganham a permissao os setores que
--  ja revisavam a despesa geral (o Envios gerais do Reembolso mudou para
--  dentro da area) ou gerenciavam o ponto de todos, e o acesso total (que
--  ja passa por todas; ela entra na lista so para a tela de Setores mostrar
--  a caixa marcada). A TV nao ganha.
--
--  Uma vez so: rodar de novo nao devolve a permissao a quem alguem tirou
--  depois (a marca 'marco_atualizacao_52' em configuracao).
--
--  So acrescenta. SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon.
-- ============================================================

BEGIN;

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['ver_administrativo']
 WHERE NOT ('ver_administrativo' = ANY (permissoes))
   AND NOT painel_tv
   AND (acesso_total
        OR 'revisar_despesa_geral' = ANY (permissoes)
        OR 'gerenciar_ponto' = ANY (permissoes))
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_52');

INSERT INTO public.configuracao (chave, valor)
VALUES ('marco_atualizacao_52', 'permissao ver_administrativo distribuida')
ON CONFLICT (chave) DO NOTHING;

COMMIT;

-- ============================================================
--  CONFERENCIA — os setores com a area
-- ============================================================

SELECT nome AS setor_com_administrativo
  FROM public.cargo
 WHERE 'ver_administrativo' = ANY (permissoes)
 ORDER BY ordem, nome;
