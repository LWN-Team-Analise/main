-- ============================================================
--  TRAJETORIA DE CLIENTES — ATUALIZACAO 4
--  Rode DEPOIS de db/atualizacao-3.sql.txt, no banco
--  "TrajetoClientes".
--
--  So ACRESCENTA. Pode rodar de novo sem problema.
--
--  O que muda:
--    1. cargo_titulo        — o cadastro de CARGOS (Analista,
--                             Coordenador...). Ate aqui o cargo era
--                             texto livre digitado em cada cadastro.
--    2. usuario.cargo_titulo_id — o vinculo com esse cadastro, para
--                             renomear um cargo valer para todo mundo.
--    3. recorte das imagens — cada foto passa a guardar o ORIGINAL e
--                             o enquadramento escolhido, e o cliente
--                             ganha um segundo recorte so para o
--                             header da obra.
-- ============================================================

BEGIN;

-- ============================================================
--  1. CADASTRO DE CARGOS
--
--  Vocabulario desta versao, porque os dois se parecem:
--
--    SETOR — o grupo da equipe (Comercial, GQ, Excelência). E ele
--            que carrega as permissoes e a cor. No banco continua
--            sendo a tabela `cargo`: renomea-la derrubaria as
--            chaves estrangeiras de meia duzia de outras tabelas.
--    CARGO — o que a pessoa e DENTRO do setor ("Analista de
--            Qualidade"). Nao decide permissao nenhuma e agora tem
--            cadastro proprio, aqui.
--
--  Cargo nao tem cor de proposito: quem pinta a tela e o setor, e
--  duas paletas competindo no mesmo card so fariam ruido.
-- ============================================================

CREATE TABLE IF NOT EXISTS public.cargo_titulo (
    id        BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome      TEXT        NOT NULL,
    ordem     INTEGER     NOT NULL DEFAULT 100,
    criado_em TIMESTAMPTZ NOT NULL DEFAULT now()
);

COMMENT ON TABLE public.cargo_titulo IS
    'Cargos da equipe (Analista, Coordenador). Sem cor e sem permissao: quem decide isso e o setor (tabela cargo).';

-- o mesmo cargo escrito com outra caixa continua sendo o mesmo cargo
CREATE UNIQUE INDEX IF NOT EXISTS cargo_titulo_nome_idx
    ON public.cargo_titulo (lower(nome));

-- ------------------------------------------------------------
--  usuario.cargo_titulo_id — o vinculo com o cadastro
--
--  A coluna de texto (usuario.cargo_titulo) FICA: e ela que umas
--  quantas telas ja leem, e apagar o que foi digitado ate hoje nao
--  traria nada. Daqui em diante as duas andam juntas — a API grava
--  o id e copia o nome.
-- ------------------------------------------------------------

ALTER TABLE public.usuario
    ADD COLUMN IF NOT EXISTS cargo_titulo_id BIGINT
        REFERENCES public.cargo_titulo(id) ON DELETE SET NULL;

COMMENT ON COLUMN public.usuario.cargo_titulo_id IS
    'Cargo da pessoa, do cadastro cargo_titulo. NULL = sem cargo.';

-- ------------------------------------------------------------
--  Os cargos que ja estavam digitados viram cadastro
--
--  Assim ninguem perde o que escreveu: cada texto distinto entra
--  como um cargo, e o usuario que o tinha ja aponta para ele.
-- ------------------------------------------------------------

INSERT INTO public.cargo_titulo (nome)
SELECT DISTINCT btrim(u.cargo_titulo)
  FROM public.usuario u
 WHERE u.cargo_titulo IS NOT NULL
   AND btrim(u.cargo_titulo) <> ''
   AND NOT EXISTS (
        SELECT 1 FROM public.cargo_titulo t
         WHERE lower(t.nome) = lower(btrim(u.cargo_titulo))
   );

UPDATE public.usuario u
   SET cargo_titulo_id = t.id
  FROM public.cargo_titulo t
 WHERE u.cargo_titulo_id IS NULL
   AND u.cargo_titulo IS NOT NULL
   AND lower(btrim(u.cargo_titulo)) = lower(t.nome);


-- ============================================================
--  2. SO CPF, NOME E SETOR SAO OBRIGATORIOS
--
--  O cadastro do colaborador passou a pedir tres coisas, e nada
--  mais: CPF, nome e setor. Data de nascimento entrou para o
--  time dos opcionais junto com e-mail, telefone, cargo e foto —
--  ela travava o cadastro de quem estava sendo registrado com o
--  documento longe da mao, e nada no sistema depende dela.
--
--  A coluna e a ultima trava que faltava: o e-mail ja tinha sido
--  liberado na atualizacao-2.
-- ============================================================

ALTER TABLE public.usuario
    ALTER COLUMN data_nascimento DROP NOT NULL;


-- ============================================================
--  3. RECORTE DAS IMAGENS
--
--  Ate aqui a foto entrava reduzida e pronto: o enquadramento era
--  o que a camera tinha decidido. Agora quem sobe a imagem escolhe
--  o que aparece — arrastando e com zoom.
--
--  Sao tres coisas guardadas, e cada uma tem um motivo:
--
--    *_original — a imagem inteira, do jeito que subiu. E dela que
--                 o editor parte quando alguem reenquadra depois;
--                 sem ela, cada ajuste recortaria o recorte
--                 anterior e a foto ia perdendo pedaco.
--    *_recorte  — { cx, cy, zoom }: onde o quadro estava. Serve
--                 para reabrir o editor no mesmo lugar.
--    a coluna da imagem em si (logo, foto, capa) continua guardando
--    o resultado JA recortado — e o que todas as telas mostram, sem
--    nenhuma delas precisar saber que existe recorte.
--
--  O cliente tem DOIS recortes da mesma imagem: o da logo (redonda,
--  no card) e o do header da obra (faixa larga). Mexer em um nao
--  mexe no outro — sao enquadramentos diferentes porque as caixas
--  tem formatos diferentes.
-- ============================================================

ALTER TABLE public.cliente
    ADD COLUMN IF NOT EXISTS logo_original TEXT,
    ADD COLUMN IF NOT EXISTS recorte_logo  JSONB,
    ADD COLUMN IF NOT EXISTS capa          TEXT,
    ADD COLUMN IF NOT EXISTS recorte_capa  JSONB;

COMMENT ON COLUMN public.cliente.logo_original IS
    'A imagem inteira que foi enviada. Base dos dois recortes.';
COMMENT ON COLUMN public.cliente.recorte_logo IS
    'Enquadramento da logo: { cx, cy, zoom }.';
COMMENT ON COLUMN public.cliente.capa IS
    'Recorte largo da mesma imagem, usado no header da obra. NULL = usa a logo.';
COMMENT ON COLUMN public.cliente.recorte_capa IS
    'Enquadramento do header: { cx, cy, zoom }.';

ALTER TABLE public.usuario
    ADD COLUMN IF NOT EXISTS foto_original TEXT,
    ADD COLUMN IF NOT EXISTS recorte_foto  JSONB;

COMMENT ON COLUMN public.usuario.foto_original IS
    'A foto inteira que foi enviada. Base do recorte do avatar.';
COMMENT ON COLUMN public.usuario.recorte_foto IS
    'Enquadramento do avatar: { cx, cy, zoom }.';

COMMIT;
