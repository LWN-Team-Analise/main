-- =====================================================================
--  1. CARGOS
--  A tela Usuarios cria, edita e apaga cargos, e escolhe a cor de cada
--  um. Os cinco cargos com fixo = true sao os que as etapas da obra
--  usam: podem mudar de nome e de cor, mas nao podem ser apagados.
--  acesso_total = true e o cargo que edita a etapa de qualquer setor
--  (diretoria).
-- =====================================================================

CREATE TABLE IF NOT EXISTS cargo (
    id            BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    -- chave e o que o codigo usa; nome e o que aparece na tela
    chave         TEXT        NOT NULL UNIQUE,
    nome          TEXT        NOT NULL,
    curto         TEXT        NOT NULL,
    cor           TEXT        NOT NULL DEFAULT '#6b7280',
    acesso_total  BOOLEAN     NOT NULL DEFAULT false,
    fixo          BOOLEAN     NOT NULL DEFAULT false,
    ordem         SMALLINT    NOT NULL DEFAULT 100,
    criado_em     TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT cargo_chave_forma CHECK (chave ~ '^[a-z][a-z0-9_]*$'),
    -- 6 digitos = cor cheia; 8 = cor com opacidade (#RRGGBBAA)
    CONSTRAINT cargo_cor_hex     CHECK (cor ~* '^#([0-9a-f]{6}|[0-9a-f]{8})$'),
    CONSTRAINT cargo_curto_tam   CHECK (char_length(curto) BETWEEN 1 AND 6)
);

COMMENT ON COLUMN cargo.chave        IS 'Identificador usado pelo sistema. Os 5 fixos sao os setores das etapas.';
COMMENT ON COLUMN cargo.curto        IS 'Etiqueta curta do card da obra: com, adm, tec, gq, exc.';
COMMENT ON COLUMN cargo.acesso_total IS 'true = edita a etapa de qualquer setor (diretoria).';
COMMENT ON COLUMN cargo.fixo         IS 'true = cargo usado pelas etapas; nao pode ser apagado.';

-- =====================================================================
--  2. USUARIO — liga na tabela cargo
--  A tabela ja existe (db/usuario.sql.txt). Aqui so acrescentamos a
--  ligacao com cargo e a foto. A coluna antiga "cargo" (texto) continua
--  existindo e e usada para preencher cargo_id automaticamente.
-- =====================================================================

ALTER TABLE usuario ADD COLUMN IF NOT EXISTS cargo_id BIGINT REFERENCES cargo (id);
ALTER TABLE usuario ADD COLUMN IF NOT EXISTS foto     TEXT;

COMMENT ON COLUMN usuario.foto IS 'Imagem em data URL ou caminho. NULL = usa as iniciais.';

-- tira acento sem depender da extensao unaccent (que nem sempre esta
-- instalada). So o que a gente usa em portugues.
CREATE OR REPLACE FUNCTION sem_acento(texto TEXT) RETURNS TEXT AS $$
    SELECT translate(
        lower(coalesce(texto, '')),
        'áàâãäéèêëíìîïóòôõöúùûüçñ',
        'aaaaaeeeeiiiiooooouuuucn'
    );
$$ LANGUAGE sql IMMUTABLE;

-- casa o texto que ja esta em usuario.cargo com o cargo certo,
-- ignorando maiuscula/minuscula e acento
UPDATE usuario u
   SET cargo_id = c.id
  FROM cargo c
 WHERE u.cargo_id IS NULL
   AND sem_acento(u.cargo) IN (c.chave, sem_acento(c.nome));

CREATE INDEX IF NOT EXISTS usuario_cargo_idx ON usuario (cargo_id);


-- =====================================================================
--  3. CLIENTE
-- =====================================================================

CREATE TABLE IF NOT EXISTS cliente (
    id         BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome       TEXT        NOT NULL,
    logo       TEXT,
    endereco   TEXT,
    bairro     TEXT,
    cidade     TEXT        NOT NULL,
    estado     CHAR(2)     NOT NULL,
    cep        CHAR(8),
    criado_em  TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT cliente_cep_digitos CHECK (cep IS NULL OR cep ~ '^[0-9]{8}$'),
    CONSTRAINT cliente_uf_forma    CHECK (estado ~ '^[A-Z]{2}$')
);

COMMENT ON COLUMN cliente.cep  IS 'Somente digitos (8). A mascara e aplicada na interface.';
COMMENT ON COLUMN cliente.logo IS 'Logo em data URL. NULL = usa as iniciais do nome.';

CREATE INDEX IF NOT EXISTS cliente_nome_idx ON cliente (lower(nome));


-- =====================================================================
--  4. OBRA
--  tipo 'padrao'      -> as 5 etapas abrem em fila (uma fecha, a outra abre)
--  tipo 'emergencia'  -> todas as etapas ja nascem liberadas
-- =====================================================================

CREATE TABLE IF NOT EXISTS obra (
    id             BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    cliente_id     BIGINT      NOT NULL REFERENCES cliente (id) ON DELETE RESTRICT,
    descricao      TEXT        NOT NULL,
    tipo           TEXT        NOT NULL DEFAULT 'padrao',
    prioridade     TEXT        NOT NULL DEFAULT 'media',
    data_prevista  DATE,
    criado_por     BIGINT      REFERENCES usuario (id),
    criado_em      TIMESTAMPTZ NOT NULL DEFAULT now(),
    atualizado_em  TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT obra_tipo_valido       CHECK (tipo IN ('padrao', 'emergencia')),
    CONSTRAINT obra_prioridade_valida CHECK (prioridade IN ('baixa', 'media', 'alta'))
);

CREATE INDEX IF NOT EXISTS obra_cliente_idx    ON obra (cliente_id);
CREATE INDEX IF NOT EXISTS obra_prioridade_idx ON obra (prioridade);


-- =====================================================================
--  5. TAREFAS DA OBRA
--  Uma linha por tarefa: qual obra, qual etapa (1..5), qual cargo
--  responde por ela, o titulo e se ja foi feita.
--  A etapa fecha quando todas as tarefas dela estao feitas.
-- =====================================================================

CREATE TABLE IF NOT EXISTS obra_tarefa (
    id           BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    obra_id      BIGINT      NOT NULL REFERENCES obra (id)  ON DELETE CASCADE,
    etapa        SMALLINT    NOT NULL,
    cargo_id     BIGINT      NOT NULL REFERENCES cargo (id) ON DELETE RESTRICT,
    ordem        SMALLINT    NOT NULL DEFAULT 0,
    titulo       TEXT        NOT NULL,
    feito        BOOLEAN     NOT NULL DEFAULT false,
    feito_por    BIGINT      REFERENCES usuario (id),
    feito_em     TIMESTAMPTZ,

    CONSTRAINT obra_tarefa_etapa_faixa CHECK (etapa BETWEEN 1 AND 5)
);

CREATE INDEX IF NOT EXISTS obra_tarefa_obra_idx  ON obra_tarefa (obra_id, etapa);
CREATE INDEX IF NOT EXISTS obra_tarefa_cargo_idx ON obra_tarefa (cargo_id);

-- carimba quem marcou e quando, sem a aplicacao precisar lembrar
CREATE OR REPLACE FUNCTION obra_tarefa_carimbo() RETURNS trigger AS $$
BEGIN
    IF NEW.feito IS DISTINCT FROM OLD.feito THEN
        NEW.feito_em := CASE WHEN NEW.feito THEN now() ELSE NULL END;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS obra_tarefa_carimbo_tg ON obra_tarefa;
CREATE TRIGGER obra_tarefa_carimbo_tg
    BEFORE UPDATE ON obra_tarefa
    FOR EACH ROW EXECUTE FUNCTION obra_tarefa_carimbo();


-- =====================================================================
--  6. QUEM PARTICIPA DA OBRA
--  Preenchida sozinha: quem marca uma tarefa entra como participante.
--  E dai que saem os avatares do card e a lista de Membros.
-- =====================================================================

CREATE TABLE IF NOT EXISTS obra_membro (
    obra_id     BIGINT      NOT NULL REFERENCES obra (id)    ON DELETE CASCADE,
    usuario_id  BIGINT      NOT NULL REFERENCES usuario (id) ON DELETE CASCADE,
    entrou_em   TIMESTAMPTZ NOT NULL DEFAULT now(),
    PRIMARY KEY (obra_id, usuario_id)
);

CREATE OR REPLACE FUNCTION obra_membro_entra() RETURNS trigger AS $$
BEGIN
    IF NEW.feito AND NEW.feito_por IS NOT NULL THEN
        INSERT INTO obra_membro (obra_id, usuario_id)
        VALUES (NEW.obra_id, NEW.feito_por)
        ON CONFLICT DO NOTHING;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS obra_membro_entra_tg ON obra_tarefa;
CREATE TRIGGER obra_membro_entra_tg
    AFTER INSERT OR UPDATE ON obra_tarefa
    FOR EACH ROW EXECUTE FUNCTION obra_membro_entra();


-- =====================================================================
--  7. OBSERVACOES DA OBRA
-- =====================================================================

CREATE TABLE IF NOT EXISTS obra_observacao (
    id          BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    obra_id     BIGINT      NOT NULL REFERENCES obra (id)    ON DELETE CASCADE,
    usuario_id  BIGINT      REFERENCES usuario (id)          ON DELETE SET NULL,
    autor_nome  TEXT        NOT NULL,
    texto       TEXT        NOT NULL,
    enviada_em  TIMESTAMPTZ NOT NULL DEFAULT now()
);

COMMENT ON COLUMN obra_observacao.autor_nome IS 'Copia do nome no momento do envio: a observacao sobrevive ao usuario sair.';

CREATE INDEX IF NOT EXISTS obra_observacao_obra_idx ON obra_observacao (obra_id, enviada_em DESC);


-- =====================================================================
--  8. AVALIACAO DA OBRA
--  Uma nota por obra, dada pela empresa cliente ao servico prestado.
--  A nota do usuario NAO fica guardada: e a media das obras em que ele
--  participou (view usuario_media abaixo).
-- =====================================================================

CREATE TABLE IF NOT EXISTS obra_avaliacao (
    obra_id     BIGINT       PRIMARY KEY REFERENCES obra (id) ON DELETE CASCADE,
    nota        NUMERIC(3,1) NOT NULL,
    descricao   TEXT,
    avaliado_por BIGINT      REFERENCES usuario (id),
    avaliado_em TIMESTAMPTZ  NOT NULL DEFAULT now(),

    CONSTRAINT obra_avaliacao_faixa CHECK (nota >= 0 AND nota <= 10)
);

COMMENT ON TABLE obra_avaliacao IS 'Nota que o cliente deu a obra. Sem linha aqui = "sem avaliacao".';

-- media de cada usuario: a nota das obras em que ele participou
CREATE OR REPLACE VIEW usuario_media AS
SELECT u.id                              AS usuario_id,
       count(a.obra_id)                  AS obras_avaliadas,
       round(avg(a.nota), 1)             AS media
  FROM usuario u
  JOIN obra_membro   m ON m.usuario_id = u.id
  JOIN obra_avaliacao a ON a.obra_id   = m.obra_id
 GROUP BY u.id;

COMMENT ON VIEW usuario_media IS 'Media das notas das obras em que o usuario participou. Sem obra avaliada = usuario nao aparece.';


-- =====================================================================
--  9. AVISOS ENVIADOS
-- =====================================================================

CREATE TABLE IF NOT EXISTS obra_aviso (
    id          BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    obra_id     BIGINT      NOT NULL REFERENCES obra (id) ON DELETE CASCADE,
    etapa       SMALLINT    NOT NULL,
    mensagem    TEXT,
    enviado_por BIGINT      REFERENCES usuario (id),
    enviado_em  TIMESTAMPTZ NOT NULL DEFAULT now()

CREATE TABLE IF NOT EXISTS obra_aviso_cargo (
    aviso_id BIGINT NOT NULL REFERENCES obra_aviso (id) ON DELETE CASCADE,
    cargo_id BIGINT NOT NULL REFERENCES cargo (id)      ON DELETE CASCADE,
    PRIMARY KEY (aviso_id, cargo_id)
);

BEGIN
    NEW.atualizado_em := now();

--  12. CONFERINDO

-- quem esta cadastrado e com qual cargo
SELECT u.id, u.name, u.cpf, u.email, u.telefone, c.nome AS cargo, c.cor
  FROM usuario u
  LEFT JOIN cargo c ON c.id = u.cargo_id
 ORDER BY c.ordem NULLS LAST, u.name;

-- media de cada usuario (so aparece quem participou de obra ja avaliada)
SELECT u.name, m.obras_avaliadas, m.media
  FROM usuario u
  JOIN usuario_media m ON m.usuario_id = u.id
 ORDER BY m.media DESC;


-- =====================================================================
--  13. QUANDO A OBRA FECHOU
--
--  A tela "Concluidas" agrupa por ano e mes. A data de conclusao nao
--  precisa de coluna nova: e o carimbo da ULTIMA tarefa marcada, que o
--  gatilho da secao 5 ja grava em obra_tarefa.feito_em.
--
--  Se voce ja rodou este arquivo antes, basta rodar SO esta secao.
-- =====================================================================

CREATE OR REPLACE VIEW obra_conclusao AS
SELECT o.id                       AS obra_id,
       max(t.feito_em)            AS concluida_em,
       count(*) FILTER (WHERE NOT t.feito) AS tarefas_abertas
  FROM obra o
  JOIN obra_tarefa t ON t.obra_id = o.id
 GROUP BY o.id
HAVING count(*) FILTER (WHERE NOT t.feito) = 0;

COMMENT ON VIEW obra_conclusao IS
    'Obras com todas as tarefas feitas. concluida_em = carimbo da ultima tarefa marcada.';

-- as concluidas, agrupadas como a tela mostra
SELECT date_part('year',  c.concluida_em)::int AS ano,
       date_part('month', c.concluida_em)::int AS mes,
       count(*)                                AS concluidas,
       count(*) FILTER (WHERE o.tipo = 'emergencia') AS emergencias,
       round(avg(a.nota), 1)                   AS nota_media
  FROM obra_conclusao c
  JOIN obra o            ON o.id = c.obra_id
  LEFT JOIN obra_avaliacao a ON a.obra_id = o.id
 GROUP BY ano, mes
 ORDER BY ano DESC, mes DESC;
