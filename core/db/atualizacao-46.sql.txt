--  ATUALIZACAO 46 — o AVISO GERAL (o "Enviar aviso" do botao +)
--
--  O botao flutuante "+" passa a existir em todas as abas. Alem do chat
--  da equipe, quem tem `enviar_avisos` ve nele o "Enviar aviso": um
--  recado para a empresa inteira (manutencao no sistema, ponto
--  facultativo, novas diretrizes), que aparece no card "Avisos e
--  Comunicados" da aba Panorama, na pagina inicial, para todo mundo.
--
--  Os avisos que ja existiam (obra_aviso) sao de UMA obra e vao para os
--  setores escolhidos; este nao tem obra nem setor:
--
--    aviso_geral   o titulo, o texto (opcional), se e importante (sai
--                  com o pingo laranja), quem mandou e quando.
--
--  Ver e de todo mundo que entra no sistema. Enviar e excluir e de quem
--  tem `enviar_avisos` (o acesso total passa por qualquer permissao).
--  Nao ha permissao nova.
--
--  SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon.
--
--  Rodar uma vez. Rodar de novo nao quebra nada e nao duplica nada.
-- ============================================================

BEGIN;

CREATE TABLE IF NOT EXISTS public.aviso_geral (
    id           bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    titulo       text        NOT NULL,
    mensagem     text        NOT NULL DEFAULT '',
    importante   boolean     NOT NULL DEFAULT false,
    enviado_por  bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    enviado_em   timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT aviso_geral_titulo_valido CHECK (btrim(titulo) <> '' AND length(titulo) <= 120),
    CONSTRAINT aviso_geral_mensagem_valida CHECK (length(mensagem) <= 2000)
);

CREATE INDEX IF NOT EXISTS aviso_geral_enviado_em_idx ON public.aviso_geral (enviado_em DESC);

COMMENT ON TABLE public.aviso_geral IS
    'Aviso para a empresa inteira (o "Enviar aviso" do botao +), no card Avisos e Comunicados do Panorama. Todos veem; quem tem enviar_avisos envia e exclui.';
COMMENT ON COLUMN public.aviso_geral.importante IS
    'true = sai em destaque (pingo laranja) no card do Panorama.';

COMMIT;
