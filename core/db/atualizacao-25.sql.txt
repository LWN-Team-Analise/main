-- ============================================================
--  ATUALIZACAO 25 — ENSAIOS POR ITEM (AMBIENTE, TAG, PONTO DE USO)
--
--  Ate aqui o Planejamento de ensaios escolhia uma lista solta para a
--  obra inteira ("esta obra faz Contagem de particulas"). Agora cada
--  ensaio fica ligado a ONDE ele e feito:
--
--    Sala 101 (ISO 7)         Contagem de particulas  As-Found
--                             Contagem de particulas  Como Certificado
--    Cabine CSB-03 (TAG)      Velocidade de face — Inflow  Como Certificado
--    Ponto de uso 12 (N2)     Pureza do gas           Como Certificado
--
--  1. obra.servico — o tipo de servico da obra inteira (um por obra):
--     qualificacao (Qualificacao/Certificacao), qi, qo, qp, tab,
--     comissionamento ou requalificacao. NULL = ainda nao escolhido.
--
--  2. obra_item — o que vai ser ensaiado na obra:
--       ambiente     sala, com a classe (ISO 7, Grau B...);
--       sistema      AHU, rede de gases...;
--       equipamento  cabine, capela, isolador — com a TAG;
--       ponto_uso    ponto de uso de gas — com o gas.
--
--  3. obra_item_ensaio — cada LINHA: item + ensaio + condicao
--     (as_found ou como_certificado). E tambem o resultado da
--     execucao dela: a data, o valor medido, se ficou conforme e a
--     observacao. Linha com data = feita.
--
--  As tabelas antigas continuam valendo: obra_ensaio passa a ser os
--  ensaios DISTINTOS das linhas, e obra_ensaio_progresso o % de cada
--  ensaio por dia (linhas feitas / linhas do ensaio). Quem le so elas
--  (o quadro, o check "Material de gases", a conclusao) nao muda.
--  As obras que ja planejaram pela lista solta continuam como estao.
--
--  SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon.
--
--  Rodar uma vez. Rodar de novo nao quebra nada e nao duplica nada.
-- ============================================================

BEGIN;

-- ---- 1. o servico da obra ----
ALTER TABLE public.obra ADD COLUMN IF NOT EXISTS servico text;

ALTER TABLE public.obra DROP CONSTRAINT IF EXISTS obra_servico_valido;
ALTER TABLE public.obra ADD CONSTRAINT obra_servico_valido
    CHECK (servico IS NULL OR servico IN ('qualificacao', 'qi', 'qo', 'qp', 'tab',
                                          'comissionamento', 'requalificacao'));

COMMENT ON COLUMN public.obra.servico IS
    'qualificacao, qi, qo, qp, tab, comissionamento ou requalificacao — um por obra. NULL = nao escolhido.';

-- ---- 2. os itens da obra ----
CREATE TABLE IF NOT EXISTS public.obra_item (
    id          bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    obra_id     bigint      NOT NULL REFERENCES public.obra (id) ON DELETE CASCADE,
    tipo        text        NOT NULL,
    nome        text        NOT NULL,
    tag         text,
    classe      text,
    gas         text,
    observacao  text,
    ordem       integer     NOT NULL DEFAULT 0,
    criado_por  bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    criado_em   timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT obra_item_tipo_valido CHECK (tipo IN ('ambiente', 'sistema', 'equipamento', 'ponto_uso')),
    CONSTRAINT obra_item_nome_cheio  CHECK (btrim(nome) <> '')
);

CREATE INDEX IF NOT EXISTS obra_item_da_obra ON public.obra_item (obra_id, ordem);

COMMENT ON TABLE public.obra_item IS
    'O que a obra vai ensaiar: ambiente (com classe), sistema, equipamento (com TAG) ou ponto de uso (com o gas).';

-- ---- 3. as linhas: item + ensaio + condicao, com o resultado ----
CREATE TABLE IF NOT EXISTS public.obra_item_ensaio (
    id             bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    item_id        bigint      NOT NULL REFERENCES public.obra_item (id) ON DELETE CASCADE,
    ensaio_id      bigint      NOT NULL REFERENCES public.ensaio (id) ON DELETE CASCADE,
    condicao       text        NOT NULL,
    ordem          integer     NOT NULL DEFAULT 0,
    data           date,
    resultado      text,
    conforme       boolean,
    observacao     text,
    registrado_por bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    registrado_em  timestamptz,
    CONSTRAINT obra_item_ensaio_condicao_valida CHECK (condicao IN ('as_found', 'como_certificado')),
    -- feita = data, resultado e conforme juntos; pendente = nenhum dos tres
    CONSTRAINT obra_item_ensaio_resultado_inteiro
        CHECK ((data IS NULL AND resultado IS NULL AND conforme IS NULL)
            OR (data IS NOT NULL AND resultado IS NOT NULL AND conforme IS NOT NULL)),
    UNIQUE (item_id, ensaio_id, condicao)
);

CREATE INDEX IF NOT EXISTS obra_item_ensaio_do_ensaio ON public.obra_item_ensaio (ensaio_id);

COMMENT ON TABLE public.obra_item_ensaio IS
    'Uma linha do planejamento: o ensaio, no item, numa condicao (as_found ou como_certificado). Com data = feita, e o resultado fica aqui.';

COMMIT;

-- ============================================================
--  CONFERENCIA — deve devolver "t" em todas as linhas
-- ============================================================

SELECT 'obra.servico' AS item,
       EXISTS (SELECT 1 FROM information_schema.columns
                WHERE table_name = 'obra' AND column_name = 'servico') AS ok
UNION ALL
SELECT 'obra_item', to_regclass('public.obra_item') IS NOT NULL
UNION ALL
SELECT 'obra_item_ensaio', to_regclass('public.obra_item_ensaio') IS NOT NULL;
