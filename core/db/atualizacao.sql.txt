BEGIN;

-- ============================================================
--  1. PERMISSOES DO CARGO
--  Cada cargo guarda a lista de chaves que ele pode. Vazio =
--  nenhuma permissao, que e o padrao pedido: quem administra
--  marca uma a uma na tela de Cargos.
-- ============================================================

ALTER TABLE public.cargo
    ADD COLUMN IF NOT EXISTS permissoes text[] NOT NULL DEFAULT '{}'::text[];

COMMENT ON COLUMN public.cargo.permissoes IS
    'Chaves de permissao do cargo (ver_obras, editar_obras, ...). Vazio = so o proprio perfil.';

-- quem tem acesso total (diretoria) continua passando por tudo,
-- mas ja nasce com a lista cheia para a tela mostrar certo
UPDATE public.cargo
   SET permissoes = ARRAY[
        'ver_inicio','ver_obras','ver_clientes','ver_concluidas','ver_avaliacoes','ver_avisos',
        'editar_usuario','editar_cargo','editar_avaliacoes','editar_clientes','editar_obras',
        'check_todas_etapas','editar_etapa','editar_cards','editar_cargos_card','editar_checks',
        'enviar_avisos'
   ]
 WHERE acesso_total AND cardinality(permissoes) = 0;


-- ============================================================
--  2. OBRA — datas, autoria e ultima alteracao
--  "data_prevista" sai de cena (a coluna fica, so para nao
--  perder o que ja estava gravado) e entram inicio/conclusao.
-- ============================================================

ALTER TABLE public.obra
    ADD COLUMN IF NOT EXISTS data_inicio date,
    ADD COLUMN IF NOT EXISTS data_conclusao date,
    ADD COLUMN IF NOT EXISTS atualizado_por bigint;

-- obras antigas: o inicio vira o dia em que foram criadas
UPDATE public.obra SET data_inicio = criado_em::date WHERE data_inicio IS NULL;

ALTER TABLE public.obra ALTER COLUMN data_inicio SET DEFAULT CURRENT_DATE;

DO $bloco$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'obra_atualizado_por_fk') THEN
        ALTER TABLE public.obra
            ADD CONSTRAINT obra_atualizado_por_fk
            FOREIGN KEY (atualizado_por) REFERENCES public.usuario(id) ON DELETE SET NULL;
    END IF;
END
$bloco$;

COMMENT ON COLUMN public.obra.data_inicio IS 'Dia em que a obra comecou. Preenchido com hoje no cadastro.';
COMMENT ON COLUMN public.obra.data_conclusao IS 'Previsao/registro de entrega. NULL = ainda sem data.';
COMMENT ON COLUMN public.obra.atualizado_por IS 'Quem fez a ultima alteracao — aparece na faixa da obra.';


-- ============================================================
--  3. ETIQUETAS
--  Rotulo livre que a obra recebe. A etiqueta e do sistema
--  (pode ser reaproveitada); obra_etiqueta e quem usa qual.
-- ============================================================

CREATE TABLE IF NOT EXISTS public.etiqueta (
    id        bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome      text NOT NULL,
    cor       text NOT NULL DEFAULT '#6b7280',
    criado_em timestamp with time zone NOT NULL DEFAULT now(),
    CONSTRAINT etiqueta_cor_hex CHECK (cor ~* '^#([0-9a-f]{6}|[0-9a-f]{8})$'),
    CONSTRAINT etiqueta_nome_tam CHECK (char_length(btrim(nome)) BETWEEN 1 AND 40)
);

CREATE UNIQUE INDEX IF NOT EXISTS etiqueta_nome_uk
    ON public.etiqueta (lower(btrim(nome)));

CREATE TABLE IF NOT EXISTS public.obra_etiqueta (
    obra_id     bigint NOT NULL REFERENCES public.obra(id)     ON DELETE CASCADE,
    etiqueta_id bigint NOT NULL REFERENCES public.etiqueta(id) ON DELETE CASCADE,
    posta_em    timestamp with time zone NOT NULL DEFAULT now(),
    PRIMARY KEY (obra_id, etiqueta_id)
);

COMMENT ON TABLE public.etiqueta IS 'Rotulos das obras. O nome e unico, sem diferenca de maiuscula.';


-- ============================================================
--  4. ANEXOS DA OBRA

    nome        text NOT NULL,

CREATE INDEX IF NOT EXISTS obra_anexo_obra_idx

COMMENT ON COLUMN public.obra_anexo.conteudo IS 'Arquivo em data URL (base64). Limite pratico: ~6 MB.';


-- ============================================================
--  5. CHAT DA OBRA
--  Uma conversa por obra. Mensagem pode responder outra
--  (responde_a), levar arquivo e mencionar pessoas.
-- ============================================================

CREATE TABLE IF NOT EXISTS public.obra_chat (
    id          bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    obra_id     bigint NOT NULL REFERENCES public.obra(id) ON DELETE CASCADE,
    usuario_id  bigint REFERENCES public.usuario(id) ON DELETE SET NULL,
    autor_nome  text NOT NULL,
    texto       text,
    responde_a  bigint REFERENCES public.obra_chat(id) ON DELETE SET NULL,
    arquivo_nome     text,
    arquivo_tipo     text,
    arquivo_conteudo text,
    enviada_em  timestamp with time zone NOT NULL DEFAULT now(),
    editada_em  timestamp with time zone,
    CONSTRAINT obra_chat_tem_conteudo
        CHECK (btrim(coalesce(texto, '')) <> '' OR arquivo_conteudo IS NOT NULL)
);

CREATE INDEX IF NOT EXISTS obra_chat_obra_idx
    ON public.obra_chat (obra_id, enviada_em);

CREATE TABLE IF NOT EXISTS public.obra_chat_mencao (
    mensagem_id bigint NOT NULL REFERENCES public.obra_chat(id) ON DELETE CASCADE,
    usuario_id  bigint NOT NULL REFERENCES public.usuario(id)   ON DELETE CASCADE,
    PRIMARY KEY (mensagem_id, usuario_id)
);

COMMENT ON TABLE public.obra_chat IS 'Conversa da obra: texto, arquivo, resposta a outra mensagem e mencoes.';
COMMENT ON COLUMN public.obra_chat.autor_nome IS 'Copia do nome no envio: a mensagem sobrevive ao usuario sair.';


-- ============================================================
--  6. ROTEIRO COM VIGENCIA
--  A regra nova: mexer no roteiro dentro de uma obra vale
--  para ELA e para as PROXIMAS — nunca para as que ja
--  passaram.
--
--  vigente_de  = a partir de qual momento a peca existe
--  vigente_ate = quando ela saiu (NULL = ainda vale)
--
--  Uma obra criada em X enxerga a peca quando
--      vigente_de <= X  e  (vigente_ate IS NULL OU X < vigente_ate)
--
--  O que ja existe recebe '-infinity': continua valendo para
--  todas as obras antigas, como sempre valeu.
-- ============================================================

ALTER TABLE public.etapa
    ADD COLUMN IF NOT EXISTS vigente_de  timestamp with time zone NOT NULL DEFAULT '-infinity',
    ADD COLUMN IF NOT EXISTS vigente_ate timestamp with time zone;

ALTER TABLE public.etapa_card
    ADD COLUMN IF NOT EXISTS vigente_de  timestamp with time zone NOT NULL DEFAULT '-infinity',
    ADD COLUMN IF NOT EXISTS vigente_ate timestamp with time zone;

ALTER TABLE public.etapa_check
    ADD COLUMN IF NOT EXISTS vigente_de  timestamp with time zone NOT NULL DEFAULT '-infinity',
    ADD COLUMN IF NOT EXISTS vigente_ate timestamp with time zone;

COMMENT ON COLUMN public.etapa.vigente_de IS
    'Momento em que a etapa passou a valer. Obra criada antes disso nao a enxerga.';
COMMENT ON COLUMN public.etapa.vigente_ate IS
    'Momento em que a etapa saiu do roteiro. Obra criada antes disso continua com ela.';

-- A view que diz quais obras fecharam precisa contar so os checks que
-- VALEM para cada obra. Sem isso, um check criado hoje reabriria uma
-- obra concluida no mes passado.
CREATE OR REPLACE VIEW public.obra_conclusao AS
 SELECT o.id AS obra_id,
        max(m.feito_em) AS concluida_em,
        count(*) FILTER (WHERE m.obra_id IS NULL) AS checks_abertos
   FROM public.obra o
   CROSS JOIN public.etapa_check ec
   JOIN public.etapa_card kd ON kd.id = ec.card_id
   JOIN public.etapa      et ON et.id = kd.etapa_id
   LEFT JOIN public.obra_check m ON m.obra_id = o.id AND m.check_id = ec.id
  WHERE ec.vigente_de <= o.criado_em
    AND (ec.vigente_ate IS NULL OR o.criado_em < ec.vigente_ate)
    AND kd.vigente_de <= o.criado_em
    AND (kd.vigente_ate IS NULL OR o.criado_em < kd.vigente_ate)
    AND et.vigente_de <= o.criado_em
    AND (et.vigente_ate IS NULL OR o.criado_em < et.vigente_ate)
  GROUP BY o.id
 HAVING count(*) FILTER (WHERE m.obra_id IS NULL) = 0
    AND count(ec.id) > 0;

COMMENT ON VIEW public.obra_conclusao IS
    'Obras sem nenhum check em aberto, contando so os checks que valiam quando a obra nasceu.';


-- ============================================================
--  7. AVALIACOES — mais de uma por obra
--  A obra pode ter a nota do diretor, a do cliente e quantas
--  mais quiserem. obra_avaliacao passa a guardar a MEDIA, e
--  quem a mantem e um gatilho — assim a view usuario_media e
--  a tela de Concluidas continuam funcionando sem mudanca.
-- ============================================================

CREATE TABLE IF NOT EXISTS public.obra_avaliacao_item (
    id           bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    obra_id      bigint NOT NULL REFERENCES public.obra(id) ON DELETE CASCADE,
    rotulo       text NOT NULL DEFAULT 'Avaliação',
    nota         numeric(3,1) NOT NULL,
    descricao    text,
    avaliado_por bigint REFERENCES public.usuario(id) ON DELETE SET NULL,
    avaliado_em  timestamp with time zone NOT NULL DEFAULT now(),
    CONSTRAINT obra_avaliacao_item_faixa CHECK (nota >= 0 AND nota <= 10)
);

CREATE INDEX IF NOT EXISTS obra_avaliacao_item_obra_idx
    ON public.obra_avaliacao_item (obra_id, avaliado_em);

COMMENT ON TABLE public.obra_avaliacao_item IS
    'Cada nota da obra (diretor, cliente, ...). A media delas vai para obra_avaliacao.';
COMMENT ON COLUMN public.obra_avaliacao_item.rotulo IS
    'De quem e a nota: "Diretor", "Cliente" ou o nome que a tela escolher.';

-- as notas que ja existiam viram o primeiro item, marcado como do cliente
INSERT INTO public.obra_avaliacao_item (obra_id, rotulo, nota, descricao, avaliado_por, avaliado_em)
SELECT a.obra_id, 'Cliente', a.nota, a.descricao, a.avaliado_por, a.avaliado_em
  FROM public.obra_avaliacao a
 WHERE NOT EXISTS (
     SELECT 1 FROM public.obra_avaliacao_item i WHERE i.obra_id = a.obra_id
 );

-- gatilho: qualquer mexida nos itens refaz a media em obra_avaliacao
CREATE OR REPLACE FUNCTION public.obra_avaliacao_media() RETURNS trigger
    LANGUAGE plpgsql
    AS $corpo$
DECLARE
    alvo bigint := coalesce(NEW.obra_id, OLD.obra_id);
    quantas integer;
BEGIN
    SELECT count(*) INTO quantas FROM obra_avaliacao_item WHERE obra_id = alvo;

    IF quantas = 0 THEN
        DELETE FROM obra_avaliacao WHERE obra_id = alvo;
    ELSE
        INSERT INTO obra_avaliacao (obra_id, nota, descricao, avaliado_por, avaliado_em)
        SELECT alvo,
               round(avg(nota), 1),
               string_agg(nullif(btrim(coalesce(descricao, '')), ''), ' | '
                          ORDER BY avaliado_em),
               (SELECT avaliado_por FROM obra_avaliacao_item
                 WHERE obra_id = alvo ORDER BY avaliado_em DESC LIMIT 1),
               max(avaliado_em)
          FROM obra_avaliacao_item
         WHERE obra_id = alvo
        ON CONFLICT (obra_id) DO UPDATE
           SET nota = excluded.nota,
               descricao = excluded.descricao,
               avaliado_por = excluded.avaliado_por,
               avaliado_em = excluded.avaliado_em;
    END IF;

    RETURN NULL;
END;
$corpo$;

DROP TRIGGER IF EXISTS obra_avaliacao_media_tg ON public.obra_avaliacao_item;
CREATE TRIGGER obra_avaliacao_media_tg
    AFTER INSERT OR UPDATE OR DELETE ON public.obra_avaliacao_item
    FOR EACH ROW EXECUTE FUNCTION public.obra_avaliacao_media();

-- ============================================================
--  10. RECUPERACAO DE SENHA
--  Codigo de 6 digitos, valido por 3 minutos. Guardamos o
--  hash do codigo, nunca o codigo em si.
-- ============================================================

CREATE TABLE IF NOT EXISTS public.senha_codigo (
    id          bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    usuario_id  bigint NOT NULL REFERENCES public.usuario(id) ON DELETE CASCADE,
    codigo_hash text NOT NULL,
    tentativas  smallint NOT NULL DEFAULT 0,
    criado_em   timestamp with time zone NOT NULL DEFAULT now(),
    expira_em   timestamp with time zone NOT NULL DEFAULT now() + interval '3 minutes',
    usado_em    timestamp with time zone
);

CREATE INDEX IF NOT EXISTS senha_codigo_usuario_idx
    ON public.senha_codigo (usuario_id, criado_em DESC);

COMMENT ON TABLE public.senha_codigo IS
    'Codigo do "esqueci minha senha". Expira em 3 minutos; so o hash e guardado.';


-- ============================================================
--  11. OUTLOOK
--  A conta corporativa que a pessoa usa para entrar. O e-mail
--  do cadastro nao muda mais pela tela — so a senha.
-- ============================================================

ALTER TABLE public.usuario
    ADD COLUMN IF NOT EXISTS outlook_email text,
    ADD COLUMN IF NOT EXISTS outlook_id    text,
    ADD COLUMN IF NOT EXISTS outlook_em    timestamp with time zone;

ALTER TABLE public.usuario ALTER COLUMN outlook SET DEFAULT false;

-- O usuario id = 1 e protegido por gatilho: qualquer UPDATE nele exige o
-- destravamento explicito. Como aqui a mexida e so trocar NULL por false
-- na coluna outlook, o destrave vale por esta transacao e nada mais.
SET LOCAL app.desbloqueio = 'on';
UPDATE public.usuario SET outlook = false WHERE outlook IS NULL;
RESET app.desbloqueio;

COMMENT ON COLUMN public.usuario.outlook IS 'true = ja vinculou a conta do Outlook e pode entrar por ela.';
COMMENT ON COLUMN public.usuario.outlook_email IS 'E-mail da conta Microsoft vinculada.';

COMMIT;
