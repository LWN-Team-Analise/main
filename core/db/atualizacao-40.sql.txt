--  ATUALIZACAO 40 — o check "Solicitar OS" na 2a etapa
--
--  O check do Almoxarife (tipo enviar_os, da atualizacao 38) sai do card
--  do Time Tecnico da 3a etapa e vai para o card do Time Tecnico da 2a —
--  o do "Material de gases" —, com o nome "Solicitar OS". (A primeira
--  versao desta atualizacao mirava o card do "Planejamento de ensaios",
--  que em producao fica na 1a etapa; a 41 conserta o que ela mudou.)
--
--  E MUDADO de lugar, e nao versionado (vigente_ate + um check novo):
--  ele existe desde 03/11 e nenhuma obra de producao chegou a usa-lo, e
--  mudando o proprio registro o que ja estiver marcado continua marcado.
--  Vale para o roteiro e para a copia das obras com checks proprios.
--
--  O resto e como era: obrigatorio como todo check da etapa; ninguem
--  marca nem desmarca na mao; a OS ligada a obra marca quando e
--  APROVADA no LWN Control e desmarca quando e excluida. Novo e o
--  clique: ele leva ao Solicitar OS do LWN Control com o cliente e a
--  obra ja escolhidos (src/pages/Obras/CardSetor.jsx).
--
--  SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon.
--
--  Rodar de novo nao faz nada.
-- ============================================================

BEGIN;

UPDATE public.etapa_check k
   SET card_id = alvo.card_id,
       titulo = 'Solicitar OS',
       ordem = alvo.proxima
  FROM (SELECT p.card_id,
               (SELECT coalesce(max(x.ordem), 0) + 1
                  FROM public.etapa_check x
                 WHERE x.card_id = p.card_id AND x.vigente_ate IS NULL AND x.obra_id IS NULL) AS proxima
          FROM public.etapa_check p
         WHERE p.tipo = 'material_gases' AND p.vigente_ate IS NULL AND p.obra_id IS NULL
         ORDER BY p.id
         LIMIT 1) alvo
 WHERE k.tipo = 'enviar_os'
   AND k.vigente_ate IS NULL
   AND k.card_id <> alvo.card_id;

COMMENT ON COLUMN public.etapa_check.tipo IS
    'comum, planejamento_ensaios, execucao_ensaios, material_gases, prazo_execucao, reserva_carro, retirada_carro ou enviar_os (o "Solicitar OS" da 2a etapa). Todos menos comum sao do sistema: nao se excluem.';

COMMIT;

SELECT 'Solicitar OS no card do Material de gases' AS item,
       EXISTS (SELECT 1
                 FROM public.etapa_check k
                 JOIN public.etapa_check p ON p.card_id = k.card_id
                                          AND p.tipo = 'material_gases'
                                          AND p.vigente_ate IS NULL AND p.obra_id IS NULL
                WHERE k.tipo = 'enviar_os' AND k.vigente_ate IS NULL AND k.obra_id IS NULL
                  AND k.titulo = 'Solicitar OS') AS ok;
