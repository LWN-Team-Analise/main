-- ============================================================
--  ATUALIZACAO 9 — HISTORICO (registro de atividades)
--
--  O "o que eu fiz" de cada pessoa: criou cliente, editou obra,
--  marcou check, enviou despesa... Aparece na pagina inicial, na
--  secao Historico, so com o que a PROPRIA pessoa fez.
--
--  Uma tabela so para o sistema inteiro, escrita por uma funcao so
--  na API (server/atividade.js). Cada rota que muda alguma coisa
--  chama essa funcao depois de dar certo — e nao ha um historico
--  por tela, cada um com o seu formato.
--
--  O registro e uma FOTO do momento: a descricao e os detalhes
--  (nome do cliente, n° da obra, valor...) sao gravados como eram
--  na hora. Diferente do resto do banco, aqui a copia e de
--  proposito: renomear o cliente depois nao reescreve o que
--  aconteceu, e apagar a obra nao apaga o fato de que ela foi
--  criada.
--
--  Rodar uma vez. Tudo e IF NOT EXISTS: rodar de novo nao quebra
--  nada e nao apaga registro nenhum.
-- ============================================================

BEGIN;

CREATE TABLE IF NOT EXISTS public.atividade (
    id            bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    usuario_id    bigint NOT NULL REFERENCES public.usuario(id) ON DELETE CASCADE,
    -- o que aconteceu, em chave: 'obra.criada', 'check.marcado', 'despesa.enviada'...
    acao          text NOT NULL,
    -- o assunto, para a cor e o filtro: obra, cliente, check, roteiro, despesa, equipe, conta...
    categoria     text NOT NULL,
    entidade_tipo text,
    entidade_id   text,
    -- a frase, pronta: "Nova obra criada"
    descricao     text NOT NULL,
    -- o resto, como era na hora: { obra, cliente, valor, ... }
    detalhes      jsonb NOT NULL DEFAULT '{}'::jsonb,
    criado_em     timestamp with time zone NOT NULL DEFAULT now(),
    CONSTRAINT atividade_acao_tam CHECK (char_length(acao) BETWEEN 1 AND 60),
    CONSTRAINT atividade_descricao_tam CHECK (char_length(descricao) BETWEEN 1 AND 300)
);

-- a unica leitura: "as minhas, da mais nova para a mais antiga"
-- (o id cresce junto com o relogio e e o corte da paginacao)
CREATE INDEX IF NOT EXISTS atividade_usuario_idx
    ON public.atividade (usuario_id, id DESC);

COMMENT ON TABLE public.atividade IS
    'Historico de acoes de cada pessoa (pagina inicial > Historico). Escrito por server/atividade.js.';
COMMENT ON COLUMN public.atividade.detalhes IS
    'Foto do momento (nomes, valores). Nao acompanha renomeacoes posteriores, de proposito.';

COMMIT;

-- ============================================================
--  CONFERENCIA — deve devolver "t"
-- ============================================================

SELECT 'atividade' AS item, to_regclass('public.atividade') IS NOT NULL AS ok;
