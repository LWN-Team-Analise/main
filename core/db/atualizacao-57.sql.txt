-- ============================================================
--  ATUALIZACAO 57 — os pedidos de insumos (o menu por area)
--
--  O menu passou a ser por area, e a Area admin ganhou os INSUMOS: cada
--  pessoa pede o material de que precisa (Seu espaco > Pedir insumos) e
--  acompanha os seus; quem atende ve os pedidos de todos (Area admin >
--  Insumos) e marca cada um como atendido ou recusado, com uma resposta.
--
--  1. A tabela insumo_pedido: um pedido de uma pessoa (o item, a
--     quantidade, uma observacao), a situacao (pendente, atendido ou
--     recusado) e a resposta de quem atendeu.
--
--  2. As permissoes (src/domain/permissoes.js):
--       ver_insumos        pedir e acompanhar os proprios pedidos (dada a
--                          TODOS os setores, menos a TV);
--       gerenciar_insumos  atender os pedidos de todos (dada ao acesso
--                          total e a quem ja revisa as despesas de todos,
--                          `revisar_despesa_geral`: o Admin e o
--                          Administrativo).
--     Quem cuida dos setores muda depois, na tela de cada um.
--
--  As permissoes entram uma vez so (a marca 'marco_atualizacao_57').
--  So acrescenta. SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon.
-- ============================================================

BEGIN;

CREATE TABLE IF NOT EXISTS public.insumo_pedido (
    id              bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    usuario_id      bigint      NOT NULL REFERENCES public.usuario (id) ON DELETE CASCADE,
    item            text        NOT NULL,
    quantidade      integer     NOT NULL DEFAULT 1,
    observacao      text,
    situacao        text        NOT NULL DEFAULT 'pendente',
    -- o recado de quem atendeu (o motivo, quando recusa)
    resposta        text,
    respondido_por  bigint      REFERENCES public.usuario (id) ON DELETE SET NULL,
    respondido_em   timestamptz,
    criado_em       timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT insumo_pedido_item_ok CHECK (char_length(btrim(item)) BETWEEN 1 AND 120),
    CONSTRAINT insumo_pedido_quantidade_ok CHECK (quantidade BETWEEN 1 AND 9999),
    CONSTRAINT insumo_pedido_observacao_ok CHECK (observacao IS NULL OR char_length(observacao) <= 500),
    CONSTRAINT insumo_pedido_situacao_ok CHECK (situacao IN ('pendente', 'atendido', 'recusado')),
    CONSTRAINT insumo_pedido_resposta_ok CHECK (resposta IS NULL OR char_length(resposta) <= 500)
);

CREATE INDEX IF NOT EXISTS insumo_pedido_do_usuario ON public.insumo_pedido (usuario_id, criado_em DESC);
CREATE INDEX IF NOT EXISTS insumo_pedido_pendentes ON public.insumo_pedido (criado_em) WHERE situacao = 'pendente';

COMMENT ON TABLE public.insumo_pedido IS
    'Pedido de insumos (material) de uma pessoa: pendente, atendido ou recusado por quem tem gerenciar_insumos.';
COMMENT ON COLUMN public.insumo_pedido.resposta IS
    'O recado de quem atendeu ou recusou o pedido (o motivo, na recusa).';

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['ver_insumos']
 WHERE NOT ('ver_insumos' = ANY (permissoes))
   AND NOT coalesce(painel_tv, false)
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_57');

UPDATE public.cargo
   SET permissoes = permissoes || ARRAY['gerenciar_insumos']
 WHERE NOT ('gerenciar_insumos' = ANY (permissoes))
   AND NOT coalesce(painel_tv, false)
   AND (acesso_total OR 'revisar_despesa_geral' = ANY (permissoes))
   AND NOT EXISTS (SELECT 1 FROM public.configuracao WHERE chave = 'marco_atualizacao_57');

INSERT INTO public.configuracao (chave, valor)
VALUES ('marco_atualizacao_57', 'ver_insumos e gerenciar_insumos distribuidas')
ON CONFLICT (chave) DO NOTHING;

COMMIT;

-- ============================================================
--  CONFERENCIA
-- ============================================================

SELECT 'insumo_pedido' AS item, to_regclass('public.insumo_pedido') IS NOT NULL AS ok
UNION ALL SELECT 'setores com ver_insumos: ' || count(*), count(*) > 0
  FROM public.cargo WHERE 'ver_insumos' = ANY (permissoes)
UNION ALL SELECT 'setores com gerenciar_insumos: ' || count(*), count(*) > 0
  FROM public.cargo WHERE 'gerenciar_insumos' = ANY (permissoes);
